\chapter{Setting and prerequisites}
\label{ch:setting}

Throughout the blueprint, $\mathbb{K}$ denotes a non-trivially normed
field — in practice $\mathbb{R}$ or $\mathbb{C}$ — and $X, Y, Z, W$ are
Banach spaces over $\mathbb{K}$. We write $\mathcal{L}(X, Y)$ for the
space of bounded linear operators $X \to Y$ with the operator norm.

\begin{definition}[Rank of a continuous linear map]
\label{def:rank}
\lean{ContinuousLinearMap.rank}
\leanok
For $S \in \mathcal{L}(X, Y)$ the \emph{rank} of $S$ is the cardinal
dimension of its range,
\[
\operatorname{rank}(S) := \dim_{\mathbb{K}} \operatorname{range}(S)
                        \in \mathrm{Card}.
\]
\end{definition}

\chapter{Axioms of an $s$-number sequence}
\label{ch:axioms}

\begin{definition}[$s$-number sequence; Pietsch axioms]
\label{def:s-number-sequence}
\lean{SNumbers.IsSNumberSequence}
\leanok
\uses{def:rank}
A family
\[
s \;=\; (s_n)_{n \in \mathbb{N}}, \qquad
s_n : \mathcal{L}(X, Y) \longrightarrow [0, \infty),
\]
indexed for all Banach pairs $(X, Y)$ over $\mathbb{K}$ is called an
\emph{$s$-number sequence} if:
\begin{description}
  \item[(S1)]\label{ax:S1} $\|S\| = s_0(S) \ge s_1(S) \ge \dots \ge 0$
        for every $S$.
  \item[(S2)]\label{ax:S2} $s_n(S + T) \le s_n(S) + \|T\|$.
  \item[(S3)]\label{ax:S3} $s_n(BSA) \le \|B\| \, s_n(S) \, \|A\|$ for
        composable bounded operators $A, B$.
  \item[(S4)]\label{ax:S4} $s_n(S) = 0$ whenever $\operatorname{rank}(S) \le n$.
  \item[(S5)]\label{ax:S5} $s_n(\mathrm{id}_{\ell_2^{n+1}}) = 1$.
\end{description}
The bundled Lean structure splits (S1) into the three predicates
\texttt{norm\_at\_zero}, \texttt{antitone}, \texttt{nonneg}.
\end{definition}

\begin{definition}[Strict $s$-number sequence]
\label{def:strict-s-number-sequence}
\lean{SNumbers.IsStrictSNumberSequence}
\leanok
\uses{def:s-number-sequence}
An $s$-number sequence is \emph{strict} if it additionally satisfies the
strengthening
\begin{description}
  \item[(S5$'$)] $s_n(\mathrm{id}_X) = 1$ for every Banach space $X$ with
        $\dim X > n$, not merely on $X = \ell_2^{n+1}$.
\end{description}
\end{definition}

\chapter{The five canonical examples}
\label{ch:examples}

We construct the five textbook $s$-number sequences. Each definition is
encoded as a \texttt{noncomputable def} returning a real number; the proof
that it satisfies the axioms (S1)–(S5) is the content of an
\texttt{IsSNumberSequence} theorem.

\section{Approximation numbers $a_n$}
\label{sec:approx}

\begin{definition}[Approximation number]
\label{def:approximation-number}
\lean{SNumbers.approximationNumber}
\leanok
\uses{def:rank}
For $S \in \mathcal{L}(X, Y)$ and $n \in \mathbb{N}$,
\[
a_n(S) := \inf \bigl\{ \|S - A\| : A \in \mathcal{L}(X, Y),
                       \operatorname{rank}(A) \le n \bigr\}.
\]
\end{definition}

\begin{theorem}[Approximation numbers form a strict $s$-number sequence]
\label{thm:approx-is-snumber}
\lean{SNumbers.isStrictSNumberSequence_approximationNumber}
\leanok
\uses{def:approximation-number, def:strict-s-number-sequence}
Let $X, Y$ be Banach spaces over $\mathbb{K}$. Then the approximation
numbers $a = (a_n)_n$ form a \emph{strict} $s$-number sequence: they
satisfy axioms (S1)--(S5) together with the strengthening (S5$'$),
namely $a_n(\mathrm{id}_X) = 1$ whenever $\dim X > n$.
\end{theorem}
\begin{proof}
\leanok
Axioms (S1)--(S4) are proved in
\texttt{SNumbers/Approximation.lean}: \texttt{approximationNumber\_nonneg},
\texttt{approximationNumber\_antitone}, \texttt{approximationNumber\_le\_norm},
\texttt{approximationNumber\_add\_le},
\texttt{approximationNumber\_comp\_comp\_le} (using
\texttt{rank\_comp\_comp\_le} and \texttt{Real.sInf\_smul\_of\_nonneg}),
and \texttt{approximationNumber\_eq\_zero\_of\_rank\_le}. The strict
normalisation (S5$'$) is \texttt{approximationNumber\_strict}:
for any $L$ with $\operatorname{rank}(L) \le n$, the range of $L$ is a
proper finite-dimensional, hence closed, subspace of $X$; Riesz's lemma
provides $x_0 \notin \operatorname{range}(L)$ with $r \|x_0\| \le \|x_0 - L
x_0\|$ for any $r < 1$, so $\|\mathrm{id} - L\| \ge \|(\mathrm{id}-L) x_0\|
/ \|x_0\| \ge r$; letting $r \to 1$ gives $\|\mathrm{id} - L\| \ge 1$,
and combined with the trivial upper bound $a_n(\mathrm{id}) \le \|\mathrm{id}\| = 1$
we obtain equality. The classical (S5) on $\ell_2^{n+1}$ specialises (S5$'$).
The argument needs neither SVD nor Auerbach.
\end{proof}

\section{Bernstein numbers $b_n$}
\label{sec:bern}

\begin{definition}[Bernstein number]
\label{def:bernstein-number}
\lean{SNumbers.bernsteinNumber}
\leanok
For $S \in \mathcal{L}(X, Y)$ and $n \in \mathbb{N}$,
\[
b_n(S) := \sup_{\dim M = n+1}
            \inf_{\substack{x \in M \\ x \neq 0}} \frac{\|Sx\|}{\|x\|}.
\]
\end{definition}

\begin{theorem}[$b_n$ is a strict $s$-number sequence]
\label{thm:bernstein-is-snumber}
\lean{SNumbers.isStrictSNumberSequence_bernsteinNumber}
\leanok
\uses{def:bernstein-number, def:strict-s-number-sequence}
Let $\mathbb{K}$ be a non-trivially normed field and $X, Y$ normed spaces
over $\mathbb{K}$. Then the Bernstein numbers $b = (b_n)_n$ form a
\emph{strict} $s$-number sequence: they satisfy (S1)--(S5) and the
strengthening (S5$'$), namely $b_n(\mathrm{id}_X) = 1$ whenever
$\dim X > n$.
\end{theorem}

The argument needs only a non-trivially normed scalar field~$\mathbb{K}$ —
no completeness, and no Riesz lemma.
\begin{proof}
\leanok
The proof is fully formalized in \texttt{SNumbers/Bernstein.lean}, with a
single nontrivial case split for (S3). Concretely:
\begin{itemize}
  \item (S1a) For 1-dimensional $M = \operatorname{span}\{v\}$ every
    nonzero $x \in M$ is $c \cdot v$, so the gain on $M$ is exactly
    $\|S v\|/\|v\|$. Taking the supremum over $v$ gives $\|S\|$.
  \item (S1b) Every $(n+2)$-dimensional $M$ contains an
    $(n+1)$-dimensional $M' \subseteq M$, and shrinking the inf-domain
    only enlarges the gain: $\operatorname{gain}(S, M) \le
    \operatorname{gain}(S, M')$.
  \item (S2) The pointwise bound
    $\|(S+T)x\|/\|x\| \le \|Sx\|/\|x\| + \|T\|$ for nonzero
    $x \in M$ pushes through the inner $\inf$, then through the outer
    $\sup$.
  \item (S3) For each $M \subseteq W$ of dimension $n+1$, split on
    whether $A$ is injective on $M$:
    \begin{itemize}
      \item If not, some nonzero $x \in M$ satisfies $A x = 0$, hence
        $(BSA)x = 0$ and the gain on $M$ is forced to $0 \le \|B\|
        \cdot b_n(S) \cdot \|A\|$.
      \item If yes, $M' := A.M \subseteq X$ has dimension $n+1$ via the
        bijection $A : M \xrightarrow{\sim} M'$, and the per-vector
        bound
        $\|B(S(Ax))\|/\|x\| \le \|B\| \cdot \|A\| \cdot
           \|S(Ax)\|/\|Ax\|$
        gives
        $\operatorname{gain}(BSA, M) \le \|B\| \cdot \|A\| \cdot
           \operatorname{gain}(S, M') \le \|B\| \cdot \|A\| \cdot b_n(S)$.
    \end{itemize}
  \item (S4) If $\operatorname{rank}\,S \le n$ then $S$ restricted to any
    $(n+1)$-dim $M$ has rank $< \dim M$, so by rank-nullity
    $M \cap \ker S \neq \{0\}$ and the gain on $M$ is $0$.
  \item (S5$'$) For every nonzero $x$, $\|x\|/\|x\| = 1$, so the gain
    on every nontrivial $M$ is exactly $1$. The constraint
    $\dim X > n$ guarantees an $(n+1)$-dim $M$ exists, hence
    $b_n(\mathrm{id}_X) = 1$. No Riesz, no completeness.
\end{itemize}
\end{proof}

\section{Gelfand numbers $c_n$}
\label{sec:gelf}

\begin{definition}[Gelfand number]
\label{def:gelfand-number}
\lean{SNumbers.gelfandNumber}
\leanok
For $S \in \mathcal{L}(X, Y)$ and $n \in \mathbb{N}$,
\[
c_n(S) := \inf \bigl\{ \|S \restriction_M\| :
            M \subseteq X \text{ closed subspace, } \operatorname{codim} M \le n
            \bigr\}.
\]
\end{definition}

\begin{theorem}[$c_n$ is a strict $s$-number sequence]
\label{thm:gelfand-is-snumber}
\lean{SNumbers.isStrictSNumberSequence_gelfandNumber}
\leanok
\uses{def:gelfand-number, def:strict-s-number-sequence}
Let $\mathbb{K}$ be a non-trivially normed field and $X, Y$ normed spaces
over $\mathbb{K}$. Then the Gelfand numbers $c = (c_n)_n$ form a
\emph{strict} $s$-number sequence: they satisfy (S1)--(S5) and the
strengthening (S5$'$), namely $c_n(\mathrm{id}_X) = 1$ whenever
$\dim X > n$.
\end{theorem}

The argument needs only a non-trivially normed scalar field~$\mathbb{K}$;
unlike its Kolmogorov dual it requires neither completeness of
$\mathbb{K}$ nor Riesz's lemma.
\begin{proof}
\leanok
The proof is fully formalized in \texttt{SNumbers/Gelfand.lean} and is
the dual of the Kolmogorov development under the
domain~$\leftrightarrow$~codomain swap. Where Kolmogorov descends $S$
through the quotient projection $\pi_V : Y \to Y / V$ for a small $V
\subseteq Y$, Gelfand restricts $S$ through the inclusion
$\iota_M : M \hookrightarrow X$ for a large (small-codimension)
$M \subseteq X$. Concretely:
\begin{itemize}
  \item (S1a) $c_0(S) = \|S\|$ via the isometry
    $\iota_\top : \top \to X$ between the subtype $(\top : \text{Submodule})$
    and $X$.
  \item (S2) Triangle inequality on $\iota_M$:
    $(S+T) \circ \iota_M = S \circ \iota_M + T \circ \iota_M$.
  \item (S3) For closed $M \subseteq X$ of codimension $\le n$ and
    $A : W \to X$, set $M' := A^{-1}(M) = M.\text{comap}\,A$. Then
    $M'$ is closed (preimage of a closed set under a continuous map)
    and the first isomorphism theorem applied to
    $\pi_M \circ A : W \to X / M$, whose kernel is exactly $M'$,
    gives $\dim(W / M') \le \dim(X / M) \le n$. Following Pietsch,
    \emph{Eigenvalues and s-numbers}, \S2.4, we then bound the
    deviation pointwise: for $w \in M'$ we have
    $A w \in M$, so
    $\|S(A w)\| \le \|S \circ \iota_M\| \cdot \|A w\| \le
    \|S \circ \iota_M\| \cdot \|A\| \cdot \|w\|$, and one further
    $\|B\|$ on the left finishes it. No corestriction, no density of
    $\|\mathbb{K}\|$.
  \item (S4) Take $M := \ker S$, which is closed because $S$ is
    continuous; the first iso $X / \ker S \simeq \operatorname{range} S$
    gives $\dim(X / \ker S) = \mathrm{rank}\,S \le n$, and
    $S \circ \iota_M = 0$.
  \item (S5$'$) The inclusion $\iota_M$ is an isometry, so
    $\|\mathrm{id}_X \circ \iota_M\| = \|\iota_M\| = 1$ for every
    nontrivial $M$. The constraint
    $\mathrm{codim}\,M \le n < \dim X$ forces $\dim M \ge 1$ via
    $\dim(X / M) + \dim M = \dim X$, so $M$ is nontrivial. No Riesz.
\end{itemize}
\end{proof}

\section{Kolmogorov numbers $d_n$}
\label{sec:kol}

\begin{definition}[Kolmogorov number]
\label{def:kolmogorov-number}
\lean{SNumbers.kolmogorovNumber}
\leanok
For $S \in \mathcal{L}(X, Y)$ and $n \in \mathbb{N}$,
\[
d_n(S) := \inf_{\substack{V \subseteq Y \\ \dim V \le n}}
            \sup_{\|x\| \le 1} \inf_{v \in V} \|Sx - v\|.
\]
\end{definition}

\begin{theorem}[$d_n$ is a strict $s$-number sequence]
\label{thm:kolmogorov-is-snumber}
\lean{SNumbers.isStrictSNumberSequence_kolmogorovNumber}
\leanok
\uses{def:kolmogorov-number, def:strict-s-number-sequence}
Let $\mathbb{K}$ be a non-trivially normed field and $X, Y$ normed spaces
over $\mathbb{K}$. Then the Kolmogorov numbers $d = (d_n)_n$ form a
\emph{strict} $s$-number sequence: they satisfy (S1)--(S5) and the
strengthening (S5$'$), namely $d_n(\mathrm{id}_X) = 1$ whenever
$\dim X > n$.
\end{theorem}

The (S5$'$) step additionally needs $\mathbb{K}$ \emph{complete} (to close
finite-dimensional subspaces via Riesz's lemma); it requires neither a
densely normed $\mathbb{K}$ nor completeness of $X$.
\begin{proof}
\leanok
The proof is fully formalized in \texttt{SNumbers/Kolmogorov.lean}, via
the operator-norm-on-quotient identity
$d_n S = \inf_V \|\pi_V \circ S\|$, where $\pi_V : Y \to Y / V$ is the
canonical projection. (S2) follows from the triangle inequality applied
to $\pi_V \circ S$ in the quotient; (S3) uses the per-subspace deviation
lemma \texttt{deviationFromSubspace\_comp\_comp\_map\_le}; (S5$'$) is
\texttt{kolmogorovNumber\_strict}, via Riesz's lemma on the closed
proper subspace $V = \mathrm{range}\,L$.
\end{proof}

\subsection{Alternative: Pietsch's lifting identity}
\label{subsec:kol-lifting}

A second formalization of the Kolmogorov numbers, restricted to the
Banach-space setting, lives in
\texttt{SNumbers/KolmogorovLifting.lean} (sub-namespace
\texttt{SNumbers.Lifting}). It uses Pietsch's identity
\[
  d_n(S) = a_n(S \circ Q_X),
\]
where $a_n$ is the approximation number and
$Q_X : \ell^1(B_X) \twoheadrightarrow X$ is the canonical summation
surjection from the $\ell^1$ space indexed by the closed unit ball
$B_X$, given by
\[
  Q_X(\alpha) := \sum_{x \in B_X} \alpha(x) \cdot x.
\]

\begin{theorem}[$d_n$ via Pietsch's lifting identity]
\label{thm:kolmogorov-lifting}
\lean{SNumbers.Lifting.kolmogorovNumber_strict,
      SNumbers.Lifting.kolmogorovNumber_comp_comp_le,
      SNumbers.Lifting.kolmogorovNumber_zero_eq_norm}
\leanok
\uses{def:kolmogorov-number, def:approximation-number}
For every Banach space $X$ and every $S \in \mathcal{L}(X, Y)$,
$d_n(S) = a_n(S \circ Q_X)$, and the Kolmogorov-number axioms
(S1)–(S5)+(S5$'$) follow from the corresponding axioms for the
approximation numbers. The (S3) ideal property factors as
$(B \circ S \circ A) \circ Q_W = B \circ (S \circ Q_X) \circ T_{A,c}$
through the lift
$T_{A,c} : \ell^1(B_W) \to \ell^1(B_X)$ defined on basis vectors by
$T_{A,c}(\delta_w) := \mathrm{lp.single}\,\langle c^{-1} A w\rangle\,c$,
with $\|T_{A,c}\| \le \|c\|$. Taking $\|c\| \searrow \|A\|$ via density
of $\|\mathbb{K}\|$ yields the bound $\|B\| \cdot d_n(S) \cdot \|A\|$.
\end{theorem}
\begin{proof}
\leanok
\uses{thm:approx-is-snumber}
In this namespace $d_n(S)$ is \emph{defined} as $a_n(S \circ Q_X)$, so each
axiom reduces to the corresponding fact for the approximation numbers
(Theorem~\ref{thm:approx-is-snumber}): \texttt{kolmogorovNumber\_zero\_eq\_norm}
gives (S1a), \texttt{kolmogorovNumber\_comp\_comp\_le} gives the ideal
property (S3) via the factorisation displayed above, and
\texttt{kolmogorovNumber\_strict} gives (S5$'$). The identification of this
$d_n$ with \texttt{SNumbers.kolmogorovNumber} of the previous section is
Pietsch's identity, formalized as
\texttt{SNumbers.kolmogorovNumber\_eq\_approx} (see below).
\end{proof}

The lifting development requires \texttt{DenselyNormedField} on
$\mathbb{K}$ (for the density-of-norm scaling) and
\texttt{CompleteSpace} on the source spaces $X$, $W$ (so $Q_X(\alpha)$
converges as a series in $X$), in addition to \texttt{CompleteSpace}
$\mathbb{K}$ for (S5$'$). The two formalizations agree on Banach spaces:
Pietsch's identity \texttt{SNumbers.kolmogorovNumber\_eq\_approx} proves
$d_n(S) = a_n(S \circ Q_X)$ for the canonical quotient number, and the
equivalence \texttt{SNumbers.Lifting.kolmogorovNumber} $=$
\texttt{SNumbers.kolmogorovNumber} (\texttt{kolmogorovNumber\_eq})
follows immediately.

\begin{theorem}[Pietsch's identity]
\label{thm:pietsch-identity}
\lean{SNumbers.kolmogorovNumber_eq_approx}
\leanok
\uses{def:kolmogorov-number, def:approximation-number}
For every Banach space $X$ and every $S \in \mathcal{L}(X, Y)$, the
canonical Kolmogorov number equals the approximation number of the lift:
$d_n(S) = a_n(S \circ Q_X)$.
\end{theorem}
\begin{proof}
\leanok
\uses{thm:kolmogorov-lifting}
The inequality $d_n(S) \le a_n(S \circ Q_X)$: for a rank-$\le n$ approximant
$L$ of $S \circ Q_X$, take $V := \operatorname{range} L$; since every unit
vector $x$ is $Q_X(\delta_x)$ and $\pi_V \circ L = 0$, one gets
$\|\pi_V \circ S\| \le \|S \circ Q_X - L\|$. The reverse inequality: for an
admissible $V$ and $\varepsilon > 0$, pick $w_x \in V$ with
$\|S x - w_x\| < \|\pi_V \circ S\| + \varepsilon$ and lift to
$L(\alpha) := \sum_x \alpha(x)\, w_x$ (a rank-$\le n$ operator into the
finite-dimensional $V$), so that $\|S \circ Q_X - L\| \le \|\pi_V \circ S\| +
\varepsilon$.
\end{proof}

\section{Hilbert numbers $h_n$}
\label{sec:hil}

For the Hilbert numbers we work over $\mathbb{R}$ and use
$L_2 := \ell_2(\mathbb{N}; \mathbb{R})$.

\begin{definition}[Hilbert number]
\label{def:hilbert-number}
\lean{SNumbers.hilbertNumber}
\leanok
For $S \in \mathcal{L}(X, Y)$ and $n \in \mathbb{N}$,
\[
h_n(S) := \sup \biggl\{ \frac{a_n(B \, S \, A)}{\|B\| \, \|A\|} :
            A : L_2 \to X, \ B : Y \to L_2, \ A \neq 0, \ B \neq 0
          \biggr\}.
\]
\end{definition}

\begin{proposition}[$h_n$ is an $s$-number sequence]
\label{prop:hilbert-is-snumber}
\lean{SNumbers.isSNumberSequence_hilbertNumber}
\leanok
\uses{def:hilbert-number, def:approximation-number, def:s-number-sequence}
Let $X, Y$ be Banach spaces. Then the Hilbert numbers $h = (h_n)_n$ form an
$s$-number sequence: they satisfy axioms (S1)--(S5). (The strict
normalisation (S5$'$) is not asserted.)
\end{proposition}
\begin{proof}
\leanok
Each axiom is a named lemma in \texttt{SNumbers/Hilbert.lean}, obtained by
pushing the corresponding property of the approximation numbers $a_n$
through the supremum defining $h_n$: nonnegativity and antitonicity (S1),
subadditivity (S2) and the ideal property (S3) are
\texttt{hilbertNumber\_nonneg}, \texttt{\_antitone}, \texttt{\_add\_le} and
\texttt{\_comp\_comp\_le}; (S4) is
\texttt{hilbertNumber\_eq\_zero\_of\_rank\_le}; and the normalisations
$h_0(S) = \|S\|$ and $h_n(\mathrm{id}) = 1$ (S5) combine the upper bound
$h_n \le \|\cdot\|$ with \texttt{norm\_le\_hilbertNumber\_zero} and
\texttt{one\_le\_hilbertNumber\_id}. All five axioms are \texttt{sorry}-free.
\end{proof}

\chapter{Pietsch's sandwich theorem}
\label{ch:sandwich}

\begin{theorem}[Hilbert-space coincidence]
\label{thm:all-eq-on-hilbert}
\lean{SNumbers.allSNumbers_eq_on_HilbertSpace}
\leanok
\uses{def:s-number-sequence, def:approximation-number, thm:scalar-factorisation}
On a Hilbert space, every $s$-number sequence agrees with the
approximation numbers: $s_n(S) = a_n(S)$ for every continuous linear map
$S$ between Hilbert spaces.
\end{theorem}
\begin{proof}
\leanok
The upper bound is Theorem~\ref{thm:sandwich}. For the lower bound
$a_n(S) \le s_n(S)$: given $c < a_n(S)$, the scalar factorisation
(Theorem~\ref{thm:scalar-factorisation}) gives contractions with
$B \circ S \circ A = c \cdot \mathrm{id}_{\ell_2^{n+1}}$; chaining (S3) with
the (S5) normalisation $s_n(\mathrm{id}_{\ell_2^{n+1}}) = 1$ and homogeneity
yields $c \le s_n(S)$. Letting $c \to a_n(S)$ gives $a_n(S) \le s_n(S)$.
\end{proof}

\begin{theorem}[Pietsch \emph{sandwich}]
\label{thm:sandwich}
\lean{SNumbers.hilbertNumber_le_sn_le_approximationNumber}
\leanok
\uses{thm:approx-is-snumber, prop:hilbert-is-snumber, thm:all-eq-on-hilbert}
For every $s$-number sequence $s$ and every operator $S$,
\[
h_n(S) \;\le\; s_n(S) \;\le\; a_n(S).
\]
\end{theorem}
\begin{proof}
\leanok
The upper bound $s_n(S) \le a_n(S)$ is
\texttt{SNumbers.sn\_le\_approximationNumber}: for any $A$ with
$\operatorname{rank}(A) \le n$ axiom (S2) yields
$s_n(S) \le s_n(A) + \|S - A\| = \|S - A\|$ by (S4); take the infimum.
The lower bound $h_n(S) \le s_n(S)$: each ratio
$a_n(B S A)/(\|B\|\|A\|)$ defining $h_n(S)$ has $B S A$ an operator
between Hilbert spaces, so the coincidence
(Theorem~\ref{thm:all-eq-on-hilbert}) gives
$a_n(B S A) = s_n(B S A) \le \|B\| s_n(S) \|A\|$ by (S3); take the
supremum. Both halves are fully formalised; the coincidence theorem the lower
half invokes rests in turn on the scalar factorisation
(Theorem~\ref{thm:scalar-factorisation}), itself proved.
\end{proof}

\chapter{Basic auxiliary results}
\label{ch:basic-results}

The companion library \texttt{BasicResults} (sibling of \texttt{SNumbers})
collects generic functional-analysis results that are independent of the
s-numbers framework: currently Auerbach's lemma, fully proved.

\begin{theorem}[Auerbach's lemma]
\label{thm:auerbach}
\lean{exists_isAuerbachBasis}
\leanok
Every finite-dimensional normed space over $\mathbb{R}$ of positive
dimension admits a basis $(e_i)$ with $\|e_i\| = 1$ whose dual coordinate
functionals also satisfy $\|e_i^*\| = 1$.
\end{theorem}
\begin{proof}
\leanok
The proof is fully formalized in \texttt{BasicResults/Auerbach.lean}.
The strategy maximises $|\det|$ on a product of unit balls, then reads off
the basis and dual functionals from the maximiser.
\end{proof}

\chapter{Singular value decomposition}
\label{ch:svd}

The singular value decomposition (Schmidt representation) of a compact
operator between Hilbert spaces is formalised in \texttt{BasicResults.SVD},
via the \emph{singular-value iteration} (repeatedly peeling off the top
singular pair). The auxiliary class of \emph{approximable} operators (those
whose approximation numbers $a_n$ tend to zero) and its identification with
the compact operators on Hilbert spaces live in the separate
\texttt{AddOns} library; this split avoids relying on any
approximation-property hypothesis on the underlying space.

\section{Approximable operators}
\label{sec:svd-approximable}

\begin{definition}[Approximable operator]
\label{def:approximable}
\lean{SVD.IsApproximable}
\leanok
\uses{def:approximation-number}
$S \in \mathcal{L}(X, Y)$ is \emph{approximable} if
$a_n(S) \xrightarrow{n \to \infty} 0$, equivalently if $S$ is the
operator-norm limit of a sequence of finite-rank operators.
\end{definition}

\begin{proposition}[Closure properties of approximable operators]
\label{prop:approximable-closure}
\lean{SVD.isApproximable_iff_existsLimit,
      SVD.isApproximable_zero,
      SVD.IsApproximable.add,
      SVD.IsApproximable.smul,
      SVD.IsApproximable.comp_comp,
      SVD.IsApproximable.of_rank_le,
      SVD.isClosed_isApproximable}
\leanok
\uses{def:approximable, def:approximation-number}
The approximable operators form a closed two-sided operator ideal: a
subspace of $\mathcal{L}(X, Y)$, closed under pre/post-composition with
bounded operators, closed in operator-norm topology, and containing all
finite-rank operators.
\end{proposition}
\begin{proof}
\leanok
Each fact is proved separately in \texttt{AddOns/Approximable.lean}. The
characterisation \texttt{isApproximable\_iff\_existsLimit} produces near-
optimal rank-$\le n$ approximants $L_n$ with $\|S - L_n\| < a_n(S) +
1/(n+1)$ via \texttt{csInf\_lt\_iff}, which feeds back to give the
existence of a finite-rank limit; the converse is squeeze on
\texttt{approximationNumber\_le\_norm\_sub}. The (S2) and (S3) axioms of
$a_n$ pass to the class via squeeze; \texttt{add} uses the
$L_{\lfloor n/2 \rfloor} + K_{\lceil n/2 \rceil}$ construction together with
\texttt{LinearMap.rank\_add\_le}.
\end{proof}

\begin{theorem}[Approximable $\Rightarrow$ compact]
\label{thm:approximable-implies-compact}
\lean{SVD.IsApproximable.isCompactOperator}
\leanok
\uses{def:approximable}
Every approximable operator $S : X \to Y$ from a normed space $X$ to a
Banach space $Y$ over a locally compact field $\mathbb{K}$ is compact.
\end{theorem}
\begin{proof}
\leanok
Each finite-rank approximant $L_n$ codrestricts to its range, a finite-
dimensional subspace of $Y$ which is locally compact since $\mathbb{K}$
is (\texttt{FiniteDimensional.proper}). The codrestricted operator is
compact (\texttt{isCompactOperator\_of\_locallyCompactSpace\_dom}), and
post-composition with the inclusion preserves compactness. Compactness
is then preserved under operator-norm limits
(\texttt{isCompactOperator\_of\_tendsto}).
\end{proof}

\section{Schmidt representation}
\label{sec:svd-schmidt}

For the rest of this chapter, $H_1, H_2$ are Hilbert spaces over
$\mathbb{K} \in \{\mathbb{R}, \mathbb{C}\}$.

\begin{theorem}[Norm attained as a singular value]
\label{thm:norm-singular-value}
\lean{SVD.IsCompactOperator.norm_isSingularValue}
\leanok
Every compact $S \in \mathcal{L}(H_1, H_2)$ attains its operator norm as a
singular value: there are unit vectors $u, v$ with $S u = \|S\| \, v$ and
$S^* v = \|S\| \, u$.
\end{theorem}
\begin{proof}
\leanok
Take a maximising sequence $x_n$ of unit vectors with $\|S x_n\| \to \|S\|$;
by compactness extract a subsequence with $S x_{n_k} \to y$, and set
$v := \|S\|^{-1} y$, $u' := S^* v$. Cauchy--Schwarz on
$\langle u', x_{n_k}\rangle = \langle v, S x_{n_k}\rangle \to \|S\|$ gives
$\|u'\| = \|S\|$; equality in Cauchy--Schwarz forces $x_{n_k} \to u$, so by
continuity $S u = y = \|S\| v$.
\end{proof}

\begin{theorem}[Singular value decomposition / Schmidt representation]
\label{thm:schmidt-representation}
\lean{SVD.IsCompactOperator.SVD}
\leanok
\uses{def:approximation-number, thm:norm-singular-value}
For every compact $S \in \mathcal{L}(H_1, H_2)$ there exist
orthonormal sequences $(u_k) \subseteq H_1$, $(v_k) \subseteq H_2$ such
that
\[
  S x \;=\; \sum_{k} a_k(S) \, \langle u_k, x \rangle \, v_k
  \qquad \text{for every } x \in H_1.
\]
The singular values are exactly the approximation numbers $a_k(S)$.
\end{theorem}
\begin{proof}
\leanok
\uses{thm:norm-singular-value}
By iterating Theorem~\ref{thm:norm-singular-value}: peel off the top
singular pair $(\sigma_0, u_0, v_0)$ of $S$, replace $S$ by
$S - \sigma_0 \langle u_0, \cdot\rangle v_0$ (which is again compact), and
recurse. The variational characterisation of each $u_k$ forces the $u_k$
(and dually the $v_k$) to be orthonormal; $\sigma_k \to 0$ follows from
compactness; and the partial sums converge to $S$ in operator norm because
the $n$-th residual has norm $\sigma_n$. Eckart--Young
(Theorem~\ref{thm:eckart-young}) identifies $\sigma_k = a_k(S)$.
\end{proof}

\begin{theorem}[Eckart--Young: optimal low-rank approximation]
\label{thm:eckart-young}
\lean{SVD.IsCompactOperator.truncation_residual_eq_approxNumber}
\leanok
\uses{def:approximation-number, thm:schmidt-representation}
For every approximable $S \in \mathcal{L}(H_1, H_2)$ and every
$n \in \mathbb{N}$ there exists $L \in \mathcal{L}(H_1, H_2)$ with
$\operatorname{rank}(L) \le n$ and $\|S - L\| = a_n(S)$.
Concretely, $L = \sum_{k < n} a_k(S) \langle u_k, \cdot \rangle v_k$ is
the truncated SVD.
\end{theorem}
\begin{proof}
\leanok
\uses{thm:schmidt-representation}
Take the truncated SVD $L = \sum_{k < n} a_k(S)\langle u_k, \cdot\rangle v_k$
from Theorem~\ref{thm:schmidt-representation}, which has rank $\le n$. The
residual $S - L = \sum_{k \ge n} a_k(S)\langle u_k, \cdot\rangle v_k$ acts on the
orthonormal $(u_k)_{k \ge n}$ with weights $a_k(S) \le a_n(S)$, so its operator
norm is exactly $a_n(S)$; no rank-$\le n$ operator does better, since $a_n(S)$
is by definition the infimum of $\|S - L'\|$ over such $L'$.
\end{proof}

\begin{theorem}[Diagonal factorisation through $\ell_2^{n+1}$]
\label{thm:schmidt-diagonal}
\lean{SVD.IsCompactOperator.diagonalFactorisation}
\leanok
\uses{def:approximation-number, thm:schmidt-representation}
For every compact $S \in \mathcal{L}(H_1, H_2)$ and every
$n \in \mathbb{N}$ there exist contractions
$A \in \mathcal{L}(\ell_2^{n+1}, H_1)$ and
$B \in \mathcal{L}(H_2, \ell_2^{n+1})$ such that, on every basis vector
$e_k$ of $\ell_2^{n+1}$,
\[
  (B \circ S \circ A)(e_k) \;=\; a_k(S) \, e_k,
  \qquad 0 \le k \le n.
\]
Equivalently, $B \circ S \circ A$ is the diagonal operator
$\operatorname{diag}(a_0(S), \dots, a_n(S))$ on $\ell_2^{n+1}$.
\end{theorem}
\begin{proof}
\leanok
\uses{thm:schmidt-representation}
Using the top $n+1$ singular pairs of Theorem~\ref{thm:schmidt-representation},
let $A \in \mathcal{L}(\ell_2^{n+1}, H_1)$ send the basis vector $e_k$ to the
right singular vector $u_k$, and let $B \in \mathcal{L}(H_2, \ell_2^{n+1})$ send
$y$ to $\sum_{k \le n} \langle v_k, y\rangle e_k$, i.e.\ project onto the span of
the left singular vectors. Both are contractions because $(u_k)$ and $(v_k)$ are
orthonormal. Then $S u_k = a_k(S)\, v_k$ gives
$(B \circ S \circ A) e_k = a_k(S)\, e_k$ for $0 \le k \le n$.
\end{proof}

This factorisation needs only the top $n+1$ singular pairs — a finite
truncation, a sibling of the full SVD rather than a corollary — but,
pinning the \emph{exact} values $a_k$, it requires $S$ compact.

\begin{theorem}[Scalar factorisation]
\label{thm:scalar-factorisation}
\lean{SVD.exists_scalar_factorisation}
\leanok
\uses{def:approximation-number}
For \emph{any} bounded $S \in \mathcal{L}(H_1, H_2)$ between Hilbert spaces
and any real $c$ with $0 \le c < a_n(S)$, there are contractions
$A \in \mathcal{L}(\ell_2^{n+1}, H_1)$, $B \in \mathcal{L}(H_2, \ell_2^{n+1})$
with $B \circ S \circ A = c \cdot \mathrm{id}_{\ell_2^{n+1}}$.
\end{theorem}
\begin{proof}
\leanok
\uses{def:approximation-number}
Since $c < a_n(S)$, no rank-$\le n$ operator approximates $S$ to within $c$, so
there is an $(n+1)$-dimensional subspace $M \subseteq H_1$ on which
$\|S x\| \ge c\,\|x\|$. Let $A$ embed $\ell_2^{n+1}$ isometrically onto $M$; then
$S \circ A$ is bounded below by $c$, so $c\,(S \circ A)^{-1}$ is a contraction on
its range, and composing it with the orthogonal projection onto that range gives
a contraction $B$ with $B \circ S \circ A = c \cdot \mathrm{id}_{\ell_2^{n+1}}$.
No compactness is used.
\end{proof}

Unlike the diagonal factorisation this holds for \emph{arbitrary} bounded
$S$ (no compactness): the strict inequality $c < a_n(S)$ needs only an
$(n+1)$-dimensional subspace on which $\|S x\| \ge c\|x\|$, which exists for
every bounded operator.

\section{Compact $\Leftrightarrow$ approximable on Hilbert spaces}
\label{sec:svd-compact-equiv}

\begin{theorem}[Compact $\Rightarrow$ approximable on Hilbert spaces]
\label{thm:compact-implies-approximable}
\lean{SVD.IsCompactOperator.isApproximable}
\leanok
\uses{def:approximable, thm:schmidt-representation}
Every compact $S \in \mathcal{L}(H_1, H_2)$ between Hilbert spaces is
approximable.
\end{theorem}
\begin{proof}
\leanok
The singular value decomposition gives singular values $\sigma_n \to 0$ which, by
Eckart–Young, equal the approximation numbers $a_n(S)$; hence $a_n(S) \to 0$.
Equivalently, the SVD truncations are finite-rank operators converging to $S$ in
operator norm.
\end{proof}

\begin{theorem}[Compact $\Leftrightarrow$ approximable on Hilbert spaces]
\label{thm:compact-iff-approximable}
\lean{SVD.isApproximable_iff_isCompactOperator}
\leanok
\uses{thm:approximable-implies-compact, thm:compact-implies-approximable}
On Hilbert spaces, $S$ is approximable iff $S$ is compact.
\end{theorem}
\begin{proof}
\leanok
\uses{thm:approximable-implies-compact, thm:compact-implies-approximable}
Two implications. The forward direction $S$ approximable $\Rightarrow$ $S$
compact is Theorem~\ref{thm:approximable-implies-compact}
(\texttt{IsApproximable.isCompactOperator}); the reverse, compact
$\Rightarrow$ approximable, is Theorem~\ref{thm:compact-implies-approximable}
(\texttt{IsCompactOperator.isApproximable}). The Lean proof packages
the two as \texttt{⟨IsApproximable.isCompactOperator, IsCompactOperator.isApproximable⟩}.
\end{proof}

The SVD for compact operators is exactly
Theorem~\ref{thm:schmidt-representation}
(\texttt{SVD.IsCompactOperator.SVD}), now that the latter is stated for
compact operators directly.

\chapter{Inequalities between \texorpdfstring{$s$}{s}-numbers}
\label{ch:inequalities}

Beyond the qualitative sandwich $h_n(S) \le s_n(S) \le a_n(S)$
(Theorem~\ref{thm:sandwich}), one can compare the individual example
sequences \emph{quantitatively}. This chapter formalises two such
comparisons for operators between Banach spaces over $\mathbb{K}$:

\begin{itemize}
  \item the \emph{forward} bound $a_n(S) \le (1+\sqrt{n})\min(c_n(S), d_n(S))$
        — the largest $s$-number exceeds the Gelfand and Kolmogorov numbers by
        at most a factor $1+\sqrt{n}$ (classical; see Pietsch, \emph{s-Numbers
        of operators in Banach spaces}, Studia Math.\ \textbf{51} (1974), and
        \emph{Eigenvalues and $s$-Numbers}, Cambridge, 1987, §2.10);
  \item the \emph{reverse} bound, the \emph{maximal difference theorem}
        $\max(c_n(S), d_n(S)) \le e\,(n+1)\, h_n(S)$ — the maximal possible
        difference between the Gelfand/Kolmogorov numbers and the smallest
        $s$-numbers, the Hilbert numbers, is a factor linear in $n$ (Ullrich,
        \emph{On bounds between $s$-numbers and widths of convex sets},
        preprint, 2026). Since
        $h_n$ is the smallest $s$-number sequence this proves the
        \emph{Mityagin--Henkin conjecture} $\max(c_n, d_n) \lesssim n\, b_n$
        up to the constant $e$.
\end{itemize}

Everything in this chapter is proved in \texttt{SNumbers/Inequalities.lean}
and \texttt{SNumbers/MaxDifference.lean}. The two classical projection theorems
that drive the forward bound (Garling--Gordon and Kadets--Snobar) are themselves
fully formalised, over any \texttt{RCLike} field, by the John's-ellipsoid
development in \texttt{BasicResults/John.lean}
(Section~\ref{sec:john-ellipsoid}), whose analytic core --- the John decomposition
of identity (Theorem~\ref{thm:john-decomposition}) --- is now proved. The whole
chapter is \texttt{sorry}-free.

\section{John's ellipsoid and the projection theorems}
\label{sec:john-ellipsoid}

The sharp constant $\sqrt{n}$ in both classical projection theorems below rests
on \emph{John's ellipsoid theorem}, which is absent from Mathlib. It is developed
here in \texttt{BasicResults/John.lean} over an arbitrary \texttt{RCLike} field
$\mathbb{K}$ (so the real and complex cases are covered at once), modelling an
inscribed ellipsoid as the image $T(B_2)$ of the Euclidean unit ball under a
linear map and maximising its volume $|\det T|$. The whole development is a
complete proof, including its classical analytic core, the John decomposition of
identity (Theorem~\ref{thm:john-decomposition}); the general-purpose ingredients
of that core (compactness of convex hulls of compact sets in finite dimension,
Hahn--Banach dominated by a seminorm, trace duality, and a Weierstrass-type
product bound) live in \texttt{BasicResults/JohnAux.lean}.

\begin{theorem}[Maximal-volume inscribed ellipsoid]
\label{thm:john-maxvolume}
\lean{John.exists_maxVolume}
\leanok
For a body seminorm $p$ on $\mathbb{K}^k$ equivalent to the Euclidean norm, among
the \emph{feasible} operators (those $T$ with $p(Tu) \le \|u\|$ for all $u$) there
is one, $T_0$, of maximal $|\det|$, and it is invertible ($\det T_0 \ne 0$).
\end{theorem}
\begin{proof}
\leanok
The feasible set is closed (each $T \mapsto p(Tu)$ is continuous) and bounded
(the lower norm bound forces $\|T\| \le 1/c$), hence compact; $|\det|$ is
continuous, so it attains a maximum. A small multiple of the identity is feasible
and has nonzero determinant, so the maximiser does too.
\end{proof}

\begin{theorem}[John decomposition of identity]
\label{thm:john-decomposition}
\lean{John.john_decomposition}
\leanok
In John position --- the identity is a maximal-volume feasible operator for the
body seminorm $q$ --- the identity is a positive combination of the rank-one
projections onto contact points:
\[
  \sum_i c_i \, \langle u_i, x\rangle \, u_i = x \quad (x \in \mathbb{K}^k),
  \qquad u_i \in \operatorname{Contact}(q),\ c_i \ge 0,\ \sum_i c_i = k,
\]
where a contact point is a unit vector $u$ with $\operatorname{re}\langle x, u\rangle
\le q(x)$ for all $x$.
\end{theorem}
\begin{proof}
\leanok
The classical core of John's theorem, from first-order optimality of the
maximal-volume ellipsoid. If $k^{-1}\mathrm{id}$ lay outside the convex hull of the
contact rank-ones --- a compact convex set of operators, whose compactness uses
that convex hulls of compact sets are compact in finite dimension --- then
geometric Hahn--Banach separation, followed by trace duality, would produce a
self-adjoint trace-zero direction $H$ with $\operatorname{re}\langle u, Hu\rangle
< 0$ at every contact point. The perturbation $(1-\rho)^{-1}(\mathrm{id} + tH)$
then stays feasible for small $t>0$ (the feasibility slack grows linearly in $t$,
using that every touching point is a contact point via a seminorm-dominated
Hahn--Banach argument), while its determinant loses only $O(t^2)$ because
$\operatorname{tr} H = 0$; so $|\det| > 1$, contradicting maximality. Hence
$k^{-1}\mathrm{id}$ lies in the hull, and Carathéodory yields the finite
combination.
\end{proof}

\begin{theorem}[Kadets--Snobar projection theorem]
\label{thm:kadets-snobar}
\lean{SNumbers.exists_projection_range_eq_of_rank_le}
\leanok
\uses{thm:john-decomposition}
Let $Y$ be a normed $\mathbb{K}$-space and $V \subseteq Y$ a subspace of dimension
$\le n$. Then there is a bounded projection $P : Y \to Y$ ($P \circ P = P$) with
$\operatorname{range} P = V$ and $\|P\| \le \sqrt{n}$.
\end{theorem}
\begin{proof}
\leanok
\uses{thm:john-decomposition}
A classical result (Kadets--Snobar 1971; Pietsch 1.5.5). Put $V$ in John position
via the maximal-volume ellipsoid: the decomposition of identity
(Theorem~\ref{thm:john-decomposition}) gives contact points $u_i$, unit vectors
$v_i = M u_i \in V$, and functionals $\varphi_i = \langle u_i, M^{-1}\cdot\rangle$
of norm $\le 1$, which Hahn--Banach extends to $g_i \in Y^*$ without increasing
the norm. Then $P = \sum_i c_i\, g_i \otimes v_i$ is the identity on $V$, hence a
projection onto $V$, and weighted Cauchy--Schwarz with $\sum_i c_i = \dim V$ gives
$\|P\| \le \sqrt{\dim V} \le \sqrt{n}$. Formalised in \texttt{John.exists\_projection}.
\end{proof}

\begin{theorem}[Garling--Gordon projection theorem, $\varepsilon$-form]
\label{thm:garling-gordon}
\lean{SNumbers.exists_projection_ker_eq_of_codim_le}
\leanok
\uses{thm:john-decomposition}
Let $X$ be a normed $\mathbb{K}$-space and $M \subseteq X$ a closed subspace of
codimension $\le n$. Then for every $\varepsilon > 0$ there is a bounded
projection $P : X \to X$ ($P \circ P = P$) with $\ker P = M$ and
$\|P\| \le \sqrt{n} + \varepsilon$.
\end{theorem}
\begin{proof}
\leanok
\uses{thm:john-decomposition}
The dual of Kadets--Snobar (Garling--Gordon 1971; Pietsch 1.7.17). Run the John
development on the finite-dimensional dual $(X/M)^*$: the contact points are
represented by vectors $w_i \in X/M$ with $\|w_i\| \le 1$ (obtained by solving
the single-dual equation $\Phi^{\mathrm{flip}} w_i = \operatorname{toDual} u_i$,
avoiding the topological double dual), each lifted to $x_i \in X$ with
$\|x_i\| < 1 + \varepsilon'$ (the quotient norm is an infimum --- the sole source
of $\varepsilon$). Then $P y = \sum_i c_i\,(\Phi u_i)(\pi y)\, x_i$ has kernel $M$
and $\|P\| \le \sqrt{\operatorname{codim} M} + \varepsilon \le \sqrt{n} +
\varepsilon$. Formalised in \texttt{John.exists\_projection\_ker}. The exact bound
$\sqrt{n}$ (removing $\varepsilon$) needs a weak-$*$ limit argument; it is not
needed here, since the applications recover the sharp constant by letting
$\varepsilon \to 0$.
\end{proof}

\section{Approximation versus Gelfand and Kolmogorov numbers}
\label{sec:ineq-forward}

\begin{theorem}[$a_n \le (1+\sqrt n)\,c_n$]
\label{thm:approx-le-gelfand}
\lean{SNumbers.approximationNumber_le_sqrt_mul_gelfandNumber}
\leanok
\uses{def:approximation-number, def:gelfand-number, thm:garling-gordon}
For $S \in \mathcal{L}(X, Y)$ between Banach spaces,
$a_n(S) \le (1 + \sqrt{n})\,c_n(S)$.
\end{theorem}
\begin{proof}
\leanok
\uses{thm:garling-gordon}
Fix a closed $M \subseteq X$ of codimension $\le n$ with $\|S|_M\|$ near
$c_n(S)$. By Garling--Gordon (Theorem~\ref{thm:garling-gordon}), for each
$\varepsilon>0$ choose a projection $P$ with $\ker P = M$ and
$\|P\| \le \sqrt n + \varepsilon$. Then $L := S \circ P$ has rank
$\le \operatorname{codim} M \le n$, and $S - L = S \circ (\mathrm{id} - P)$ has
range inside $M$ (since $\operatorname{range}(\mathrm{id}-P) = \ker P = M$), so
$\|S - L\| \le \|S|_M\|\,\|\mathrm{id} - P\| \le (1+\sqrt n+\varepsilon)\,\|S|_M\|$.
Letting $\varepsilon \to 0$ gives $a_n(S) \le (1+\sqrt n)\,\|S|_M\|$; taking the
infimum over $M$ gives the claim. The per-subspace step is
\texttt{approximationNumber\_le\_sqrt\_mul\_deviationFromRestriction}, passed
to the infimum defining $c_n$.
\end{proof}

\begin{theorem}[$a_n \le (1+\sqrt n)\,d_n$]
\label{thm:approx-le-kolmogorov}
\lean{SNumbers.approximationNumber_le_sqrt_mul_kolmogorovNumber}
\leanok
\uses{def:approximation-number, def:kolmogorov-number, thm:kadets-snobar}
For $S \in \mathcal{L}(X, Y)$ between Banach spaces,
$a_n(S) \le (1 + \sqrt{n})\,d_n(S)$.
\end{theorem}
\begin{proof}
\leanok
\uses{thm:kadets-snobar}
Dual to the Gelfand half. Fix $V \subseteq Y$ of dimension $\le n$ with
$\|\pi_V \circ S\|$ near $d_n(S)$; by Kadets--Snobar
(Theorem~\ref{thm:kadets-snobar}) choose a projection $P$ onto $V$ with
$\|P\| \le \sqrt n$. Then $L := P \circ S$ has rank $\le n$, and
$S - L = (\mathrm{id} - P) \circ S$ factors through the quotient $Y / V$
(because $\mathrm{id} - P$ vanishes on $V$), so
$\|S - L\| \le \|\mathrm{id} - P\|\,\|\pi_V \circ S\| \le (1+\sqrt n)\,d_n(S)$
after taking the infimum. The per-subspace step is
\texttt{approximationNumber\_le\_sqrt\_mul\_deviationFromSubspace}.
\end{proof}

\begin{theorem}[Combined bound $a_n \le (1+\sqrt n)\min(c_n, d_n)$]
\label{thm:approx-le-min}
\lean{SNumbers.approximationNumber_le_sqrt_mul_min}
\leanok
\uses{thm:approx-le-gelfand, thm:approx-le-kolmogorov}
For $S \in \mathcal{L}(X, Y)$ between Banach spaces,
$a_n(S) \le (1 + \sqrt{n})\,\min\bigl(c_n(S), d_n(S)\bigr)$.
\end{theorem}
\begin{proof}
\leanok
\uses{thm:approx-le-gelfand, thm:approx-le-kolmogorov}
Immediate from the two halves (Theorems~\ref{thm:approx-le-gelfand}
and~\ref{thm:approx-le-kolmogorov}) by case analysis on which of $c_n(S)$,
$d_n(S)$ is the smaller.
\end{proof}

\section{The maximal difference theorem}
\label{sec:ineq-maxdiff}

The maximal difference theorem is \emph{pointwise} in the Hilbert numbers.
It is driven by the \emph{determinant quantities}
\[
  \Delta_k(S) \;:=\; \sup\bigl\{\, \|\det(B \circ S \circ A)\| \;:\;
    A : \ell_2^k \to X,\ B : Y \to \ell_2^k,\ \|A\| \le 1,\ \|B\| \le 1 \,\bigr\},
\]
formalised, together with the theorem itself, in
\texttt{SNumbers/MaxDifference.lean}. Two features make the
$\Delta_k$ work: they are closed under contractive compressions, and they do
not decay too fast from $\Delta_k$ to $\Delta_{k+1}$ (the growth lemma below).

\begin{definition}[Determinant quantities]
\label{def:det-quantity}
\lean{SNumbers.detNumber}
\leanok
For $S \in \mathcal{L}(X, Y)$ and $k \in \mathbb{N}$, $\Delta_k(S)$ is the
supremum above; here $B \circ S \circ A$ is an endomorphism of the
$k$-dimensional Hilbert space $\ell_2^k$, so its determinant is a scalar.
One has $\Delta_0(S) = 1$ (the determinant on the trivial space is $1$) and
$\Delta_k(S) \le \|S\|^k$, via $\prod_i a_i(T) = \|\det T\|$
(Lemma~\ref{lem:prod-approx-eq-det}) and $a_i(T) \le \|T\| \le \|S\|$.
\end{definition}

\begin{lemma}[$\prod_k a_k(T) = \|\det T\|$]
\label{lem:prod-approx-eq-det}
\lean{SNumbers.prod_approximationNumber_eq_norm_det}
\leanok
\uses{def:approximation-number}
For an endomorphism $T$ of a finite-dimensional inner-product space,
$\prod_{k < \dim} a_k(T) = \|\det T\|$. Indeed the approximation numbers equal
the singular values $\sigma_k(T)$ in finite dimension, and
$\|\det T\| = \prod_k \sigma_k(T)$.
\end{lemma}
\begin{proof}
\leanok
The approximation numbers coincide with Mathlib's eigenvalue-defined singular
values (\texttt{sn\_eq\_singularValues\_of\_finiteDimensional}), and
$\|\det T\| = \prod_k \sigma_k$ is pure linear algebra
(\texttt{LinearMap.norm\_det\_eq\_prod\_singularValues}, via
$\|\det T\|^2 = \det(T^*T) = \prod \lambda_k(T^*T) = \prod \sigma_k^2$).
\end{proof}

\begin{lemma}[Extension lemma]
\label{lem:det-extension}
\lean{SNumbers.exists_det_extension}
\leanok
\uses{def:det-quantity}
Let $(A, B)$ be admissible for $\Delta_k(S)$, let $u \in B_X$,
$\rho \in B_{Y^*}$, and $\lambda, \mu \ge 0$ with $\lambda^2 + \mu^2 = 1$.
If either $B(Su) = 0$ or $\rho(S(A e_j)) = 0$ for all $j$, then there is an
admissible pair $(A', B')$ for $\Delta_{k+1}(S)$ with
\[
  \|\det(B' \circ S \circ A')\| \;=\;
  \lambda^{2k} \mu^2\, \|\rho(Su)\|\, \|\det(B \circ S \circ A)\|.
\]
\end{lemma}
\begin{proof}
\leanok
Take $A'(s \oplus t) := \lambda\, A s + \mu\, t\, u$ and
$B'y := (\lambda\, B y) \oplus (\mu\, \rho(y))$. Contractivity of $A'$ is the
two-term Cauchy--Schwarz inequality
$\lambda a + \mu b \le \sqrt{a^2 + b^2}$, and
$\|B'y\|^2 = \lambda^2\|By\|^2 + \mu^2|\rho(y)|^2 \le \|y\|^2$. In the
standard basis the matrix of $B'SA'$ is the old matrix scaled by $\lambda^2$,
bordered by one row and one column of which (by hypothesis) one vanishes
outside the corner entry $\mu^2 \rho(Su)$; Laplace expansion along that line
gives the determinant identity.
\end{proof}

\begin{lemma}[Growth lemma]
\label{lem:det-growth}
\lean{SNumbers.mul_detNumber_le_detNumber_succ}
\leanok
\uses{def:det-quantity, lem:det-extension, def:gelfand-number, def:kolmogorov-number}
Let $0 \le \gamma$ with $\gamma < c_n(S)$ or $\gamma < d_n(S)$. Then for every
$k \le n$,
\[
  \frac{k^k}{(k+1)^{k+1}}\; \gamma\; \Delta_k(S) \;\le\; \Delta_{k+1}(S).
\]
In particular $\Delta_k(S) > 0$ for all $k \le n+1$ when $\gamma > 0$.
\end{lemma}
\begin{proof}
\leanok
\uses{lem:det-extension}
For each admissible pair $(A, B)$ for $\Delta_k$, an oracle provides the input
of the extension lemma. If $\gamma < c_n(S)$: the common kernel of the rows of
$B$ composed with $S$ is closed of codimension $\le k \le n$, so it contains
$u \in B_X$ with $B(Su) = 0$ and $\|Su\| > \gamma$, and Hahn--Banach provides
a norming functional $\rho$ (\texttt{exists\_gelfand\_pair}). If
$\gamma < d_n(S)$: some $u \in B_X$ has $Su$ at distance $> \gamma$ from
$V := \operatorname{span}\{S(A e_j)\}$ (dimension $\le k \le n$), and a
functional vanishing on $V$ realises that distance
(\texttt{exists\_kolmogorov\_pair}). The weights $\lambda^2 = k/(k+1)$,
$\mu^2 = 1/(k+1)$ maximise $\lambda^{2k}\mu^2$, giving exactly
$k^k/(k+1)^{k+1}$; taking the supremum over $(A, B)$ finishes. Note that no
near-maximiser has to be selected — \emph{every} admissible pair is improved,
so no $\varepsilon \to 0$ passage is needed here.
\end{proof}

\begin{lemma}[Upper bound $\Delta_{n+1} \le h_n \Delta_n$]
\label{lem:det-upper}
\lean{SNumbers.detNumber_succ_le_hilbertNumber_mul}
\leanok
\uses{def:det-quantity, lem:prod-approx-eq-det, thm:schmidt-diagonal, def:hilbert-number}
$\Delta_{n+1}(S) \le h_n(S)\, \Delta_n(S)$.
\end{lemma}
\begin{proof}
\leanok
\uses{lem:prod-approx-eq-det, thm:schmidt-diagonal}
For admissible $(A, B)$ and $T := B S A$ on $\ell_2^{n+1}$,
$\|\det T\| = \prod_{k \le n} a_k(T)$ (Lemma~\ref{lem:prod-approx-eq-det}).
The last factor satisfies $a_n(T) \le \|B\|\|A\|\, h_n(S) \le h_n(S)$. For the
top $n$ factors, the diagonal factorisation
$B_1 T A_1 = \operatorname{diag}(a_0(T), \dots, a_n(T))$ through contractions
(Theorem~\ref{thm:schmidt-diagonal}), compressed to the first $n$ coordinates
by the coordinate embedding/projection, is itself admissible for
$\Delta_n(S)$, whence $\prod_{k < n} a_k(T) \le \Delta_n(S)$. Again every
admissible pair is bounded, so the bound passes to the supremum
$\Delta_{n+1}(S)$.
\end{proof}

\begin{theorem}[Maximal difference theorem]
\label{thm:maximal-difference}
\lean{SNumbers.max_gelfandNumber_kolmogorovNumber_le_e_mul_hilbertNumber}
\leanok
\uses{lem:det-growth, lem:det-upper, def:hilbert-number}
For every $S \in \mathcal{L}(X, Y)$ and $n \in \mathbb{N}_0$,
\[
  \max\bigl(c_n(S), d_n(S)\bigr) \;\le\; \frac{(n+1)^{n+1}}{n^n}\, h_n(S)
  \;\le\; e\,(n+1)\, h_n(S).
\]
\end{theorem}
\begin{proof}
\leanok
\uses{lem:det-growth, lem:det-upper}
For $0 < \gamma < c_n(S)$ (resp.\ $< d_n(S)$), chain the growth lemma at
$k = n$ with the upper bound:
$\frac{n^n}{(n+1)^{n+1}}\, \gamma\, \Delta_n(S) \le \Delta_{n+1}(S) \le
h_n(S)\, \Delta_n(S)$, and cancel $\Delta_n(S) > 0$. Letting $\gamma$ tend to
$c_n(S)$ (resp.\ $d_n(S)$) gives the sharp constant; finally
$(n+1)^{n+1}/n^n = (n+1)(1 + 1/n)^n \le e\,(n+1)$ from $1 + x \le e^x$ at
$x = 1/n$.
\end{proof}

\begin{corollary}[Mityagin--Henkin up to the constant $e$]
\label{cor:mityagin-henkin}
\lean{SNumbers.max_gelfandNumber_kolmogorovNumber_le_e_mul_bernsteinNumber}
\leanok
\uses{thm:maximal-difference, thm:sandwich, def:bernstein-number}
$\max(c_n(S), d_n(S)) \le e\,(n+1)\, s_n(S)$ for every $s$-number sequence
$s$; in particular $\max(c_n(S), d_n(S)) \le e\,(n+1)\, b_n(S)$ for the
Bernstein numbers.
\end{corollary}
\begin{proof}
\leanok
\uses{thm:maximal-difference}
The Hilbert numbers are the smallest $s$-number sequence
(\texttt{hilbertNumber\_le\_sn}, part of Theorem~\ref{thm:sandwich}).
\end{proof}

\begin{remark}
By telescoping, $\prod_{k=0}^n (k+1)^{k+1}/k^k = (n+1)^{n+1}$,
Theorem~\ref{thm:maximal-difference} recovers the previous geometric-mean
bound $\max(c_n, d_n) \le (n+1)\,(\prod_{k=0}^n h_k)^{1/(n+1)}$
(Pietsch, \emph{Operator Ideals}; see also Ullrich, arXiv:2405.05509).
\end{remark}

\chapter{Injective and surjective \texorpdfstring{$s$}{s}-numbers}
\label{ch:injectivity}

Two further structural properties of an $s$-number sequence, from Pietsch's
theory, single out the Gelfand and Kolmogorov numbers. They concern the
behaviour of $s_n$ under composition with two distinguished classes of maps.
Everything in this chapter is fully formalised in
\texttt{SNumbers/Injectivity.lean}.

\begin{definition}[Metric injection]
\label{def:metric-injection}
\lean{SNumbers.IsMetricInjection}
\leanok
A \emph{metric injection} is a continuous linear map $J : Y \to Z$ that
preserves norms: $\|J y\| = \|y\|$ for all $y$ (an isometric embedding). In
particular $\|J\| \le 1$.
\end{definition}

\begin{definition}[Metric surjection]
\label{def:metric-surjection}
\lean{SNumbers.IsMetricSurjection}
\leanok
A \emph{metric surjection} is a continuous linear map $Q : W \to X$ that is a
contraction ($\|Q\| \le 1$) and maps the open unit ball \emph{onto} the open
unit ball: every $x$ has preimages of norm arbitrarily close to $\|x\|$. This
is the quotient-map condition $\|x\| = \inf\{\|w\| : Q w = x\}$.
\end{definition}

\begin{lemma}[Norms are preserved under the two compositions]
\label{lem:metric-norm-comp}
\lean{SNumbers.IsMetricInjection.norm_comp,
      SNumbers.IsMetricSurjection.norm_comp}
\leanok
\uses{def:metric-injection, def:metric-surjection}
Post-composition with a metric injection and pre-composition with a metric
surjection both leave the operator norm unchanged:
$\|J \circ T\| = \|T\|$ and $\|T \circ Q\| = \|T\|$.
\end{lemma}
\begin{proof}
\leanok
For the injection, $\|(J \circ T)x\| = \|J(Tx)\| = \|Tx\|$ pointwise, so the
two operator norms agree. For the surjection, $\|T \circ Q\| \le \|T\|$ since
$\|Q\| \le 1$; conversely, lifting each $x$ to a preimage $w$ with
$\|w\| \approx \|x\|$ gives $\|Tx\| = \|(T\circ Q)w\| \le \|T\circ Q\|\,\|w\|$,
and letting the slack tend to $0$ yields $\|T\| \le \|T \circ Q\|$.
\end{proof}

\begin{definition}[Injective / surjective $s$-number sequence]
\label{def:injective-surjective}
\lean{SNumbers.Injective, SNumbers.Surjective}
\leanok
\uses{def:s-number-sequence, def:metric-injection, def:metric-surjection}
An $s$-number sequence $s$ is \emph{injective} if post-composition with any
metric injection leaves it unchanged, $s_n(J \circ S) = s_n(S)$; it is
\emph{surjective} if pre-composition with any metric surjection leaves it
unchanged, $s_n(S \circ Q) = s_n(S)$.
\end{definition}

\begin{theorem}[The Gelfand numbers are injective]
\label{thm:gelfand-injective}
\lean{SNumbers.injective_gelfandNumber}
\leanok
\uses{def:injective-surjective, def:gelfand-number, lem:metric-norm-comp}
The Gelfand numbers $c_n$ form an injective $s$-number sequence:
$c_n(J \circ S) = c_n(S)$ for every metric injection $J$.
\end{theorem}
\begin{proof}
\leanok
\uses{lem:metric-norm-comp}
The infimum defining $c_n$ ranges over restriction norms $\|S|_M\|$, and
$\|(J \circ S)|_M\| = \|J \circ (S|_M)\| = \|S|_M\|$ by
Lemma~\ref{lem:metric-norm-comp}. The two infimum sets therefore coincide
term by term, so the Gelfand numbers agree.
\end{proof}

\begin{theorem}[The Kolmogorov numbers are surjective]
\label{thm:kolmogorov-surjective}
\lean{SNumbers.surjective_kolmogorovNumber}
\leanok
\uses{def:injective-surjective, def:kolmogorov-number, lem:metric-norm-comp}
The Kolmogorov numbers $d_n$ form a surjective $s$-number sequence:
$d_n(S \circ Q) = d_n(S)$ for every metric surjection $Q$.
\end{theorem}
\begin{proof}
\leanok
\uses{lem:metric-norm-comp}
The infimum defining $d_n$ ranges over quotient norms
$\|\pi_V \circ S\|$, and
$\|\pi_V \circ (S \circ Q)\| = \|(\pi_V \circ S) \circ Q\| = \|\pi_V \circ S\|$
by Lemma~\ref{lem:metric-norm-comp}. The two infimum sets coincide term by
term, so the Kolmogorov numbers agree.
\end{proof}
