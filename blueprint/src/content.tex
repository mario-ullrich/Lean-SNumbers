\chapter{Setting and prerequisites}
\label{ch:setting}

Throughout the blueprint, $\mathbb{K}$ denotes a non-trivially normed
field --- in practice $\mathbb{R}$ or $\mathbb{C}$ --- and $X, Y, Z, W$ are
Banach spaces over $\mathbb{K}$. We write $\mathcal{L}(X, Y)$ for the
space of bounded linear operators $X \to Y$ with the operator norm, and
$\mathbb{N}_0 = \{0, 1, 2, \dots\}$; all sequences are indexed starting
from $0$.

\begin{definition}[Rank of a continuous linear map]
\label{def:rank}
\lean{ContinuousLinearMap.rank}
\leanok
For $S \in \mathcal{L}(X, Y)$ the \emph{rank} of $S$ is the cardinal
dimension of its range,
\[
\operatorname{rank}(S) := \dim_{\mathbb{K}} \operatorname{range}(S)
                        \in \mathrm{Card}.
\]
\end{definition}

\chapter{Axioms of an $s$-number sequence}
\label{ch:axioms}

\begin{definition}[$s$-number sequence; Pietsch axioms]
\label{def:s-number-sequence}
\lean{SNumbers.IsSNumberSequence}
\leanok
\uses{def:rank}
A family
\[
s \;=\; (s_n)_{n \in \mathbb{N}_0}, \qquad
s_n : \mathcal{L}(X, Y) \longrightarrow [0, \infty),
\]
defined for all pairs of Banach spaces $(X, Y)$ over $\mathbb{K}$ is called an
\emph{$s$-number sequence} if:
\begin{description}
  \item[(S1)]\label{ax:S1} $\|S\| = s_0(S) \ge s_1(S) \ge \dots \ge 0$
        for every $S$.
  \item[(S2)]\label{ax:S2} $s_n(S + T) \le s_n(S) + \|T\|$.
  \item[(S3)]\label{ax:S3} $s_n(BSA) \le \|B\| \, s_n(S) \, \|A\|$ for
        composable bounded operators $A, B$.
  \item[(S4)]\label{ax:S4} $s_n(S) = 0$ whenever $\operatorname{rank}(S) \le n$.
  \item[(S5)]\label{ax:S5} $s_n(\mathrm{id}_{\ell_2^{n+1}}) = 1$.
\end{description}
\end{definition}

\begin{definition}[Strict $s$-number sequence]
\label{def:strict-s-number-sequence}
\lean{SNumbers.IsStrictSNumberSequence}
\leanok
\uses{def:s-number-sequence}
An $s$-number sequence is \emph{strict} if it additionally satisfies the
strengthening
\begin{description}
  \item[(S5$'$)] $s_n(\mathrm{id}_X) = 1$ for every Banach space $X$ with
        $\dim X > n$, not merely on $X = \ell_2^{n+1}$.
\end{description}
\end{definition}

\chapter{The five canonical examples}
\label{ch:examples}

We construct the five classical $s$-number sequences and verify the
axioms (S1)--(S5) --- and, where it holds, the strengthening (S5$'$) ---
for each of them.

\section{Approximation numbers $a_n$}
\label{sec:approx}

\begin{definition}[Approximation number]
\label{def:approximation-number}
\lean{SNumbers.approximationNumber}
\leanok
\uses{def:rank}
For $S \in \mathcal{L}(X, Y)$ and $n \in \mathbb{N}_0$,
\[
a_n(S) := \inf \bigl\{ \|S - A\| : A \in \mathcal{L}(X, Y),
                       \operatorname{rank}(A) \le n \bigr\}.
\]
\end{definition}

\begin{theorem}[Approximation numbers form a strict $s$-number sequence]
\label{thm:approx-is-snumber}
\lean{SNumbers.isStrictSNumberSequence_approximationNumber}
\leanok
\uses{def:approximation-number, def:strict-s-number-sequence}
Let $X, Y$ be Banach spaces over $\mathbb{K}$. Then the approximation
numbers $a = (a_n)_n$ form a \emph{strict} $s$-number sequence: they
satisfy the axioms (S1)--(S5) and the strengthening (S5$'$).
\end{theorem}
\begin{proof}
\leanok
Axioms (S1)--(S4) follow directly from the definition of $a_n$ as an
infimum: monotonicity and nonnegativity are inherited from the underlying
set of approximants, subadditivity (S2) uses that $A$ approximates $S + T$
whenever it approximates $S$ up to $\|T\|$, the ideal property (S3) uses
$\operatorname{rank}(B A' A) \le \operatorname{rank}(A')$, and (S4) is
witnessed by the approximant $A = S$. For (S5$'$) let $L$ have
$\operatorname{rank}(L) \le n < \dim X$; then $\operatorname{range}(L)$
is a proper finite-dimensional, hence closed, subspace of $X$, and Riesz's
lemma provides, for every $r < 1$, a unit vector $x_0$ with
$\|x_0 - L x_0\| \ge r$, so $\|\mathrm{id} - L\| \ge r$; letting
$r \to 1$ and combining with $a_n(\mathrm{id}) \le \|\mathrm{id}\| = 1$
gives $a_n(\mathrm{id}_X) = 1$. The classical (S5) on $\ell_2^{n+1}$ is the
special case $X = \ell_2^{n+1}$. The argument needs neither the SVD nor
Auerbach's lemma; it is formalised in \texttt{Approximation.lean}.
\end{proof}

\section{Bernstein numbers $b_n$}
\label{sec:bern}

\begin{definition}[Bernstein number]
\label{def:bernstein-number}
\lean{SNumbers.bernsteinNumber}
\leanok
For $S \in \mathcal{L}(X, Y)$ and $n \in \mathbb{N}_0$,
\[
b_n(S) := \sup_{\dim M = n+1}
            \inf_{\substack{x \in M \\ x \neq 0}} \frac{\|Sx\|}{\|x\|}.
\]
\end{definition}

\begin{theorem}[$b_n$ is a strict $s$-number sequence]
\label{thm:bernstein-is-snumber}
\lean{SNumbers.isStrictSNumberSequence_bernsteinNumber}
\leanok
\uses{def:bernstein-number, def:strict-s-number-sequence}
Let $\mathbb{K}$ be a non-trivially normed field and $X, Y$ normed spaces
over $\mathbb{K}$. Then the Bernstein numbers $b = (b_n)_n$ form a
\emph{strict} $s$-number sequence: they satisfy (S1)--(S5) and the
strengthening (S5$'$).
\end{theorem}

The argument needs only a non-trivially normed scalar field~$\mathbb{K}$ ---
no completeness, and no Riesz lemma.
\begin{proof}
\leanok
Write $\gamma(S, M) := \inf_{0 \ne x \in M} \|Sx\|/\|x\|$ for the gain of
$S$ on a subspace $M$, so that $b_n(S) = \sup_{\dim M = n+1} \gamma(S, M)$.
On a one-dimensional $M = \operatorname{span}\{v\}$ the gain is exactly
$\|Sv\|/\|v\|$, and the supremum over $v$ gives $b_0(S) = \|S\|$;
monotonicity in $n$ holds because every $(n+2)$-dimensional subspace
contains an $(n+1)$-dimensional one, and shrinking $M$ only enlarges the
gain. Subadditivity (S2) is the pointwise triangle inequality
$\|(S+T)x\|/\|x\| \le \|Sx\|/\|x\| + \|T\|$, pushed through the infimum and
the supremum. For the ideal property (S3) fix an $(n+1)$-dimensional
$M \subseteq W$: if $A$ is not injective on $M$, the gain of $BSA$ on $M$
vanishes; if it is, then $M' := A(M)$ has dimension $n+1$ and
$\gamma(BSA, M) \le \|B\| \, \gamma(S, M') \, \|A\| \le
\|B\| \, b_n(S) \, \|A\|$. For the rank axiom (S4), an operator of rank
$\le n$ has nontrivial kernel on every $(n+1)$-dimensional subspace by
rank--nullity, forcing the gain to zero. Finally (S5$'$): the identity has
gain $1$ on every nontrivial subspace, and $\dim X > n$ guarantees an
$(n+1)$-dimensional subspace exists. Formalised in
\texttt{Bernstein.lean}.
\end{proof}

\section{Gelfand numbers $c_n$}
\label{sec:gelf}

\begin{definition}[Gelfand number]
\label{def:gelfand-number}
\lean{SNumbers.gelfandNumber}
\leanok
For $S \in \mathcal{L}(X, Y)$ and $n \in \mathbb{N}_0$,
\[
c_n(S) := \inf \bigl\{ \|S \restriction_M\| :
            M \subseteq X \text{ closed subspace, } \operatorname{codim} M \le n
            \bigr\}.
\]
\end{definition}

\begin{theorem}[$c_n$ is a strict $s$-number sequence]
\label{thm:gelfand-is-snumber}
\lean{SNumbers.isStrictSNumberSequence_gelfandNumber}
\leanok
\uses{def:gelfand-number, def:strict-s-number-sequence}
Let $\mathbb{K}$ be a non-trivially normed field and $X, Y$ normed spaces
over $\mathbb{K}$. Then the Gelfand numbers $c = (c_n)_n$ form a
\emph{strict} $s$-number sequence: they satisfy (S1)--(S5) and the
strengthening (S5$'$).
\end{theorem}

The argument needs only a non-trivially normed scalar field~$\mathbb{K}$;
unlike its Kolmogorov dual it requires neither completeness of
$\mathbb{K}$ nor Riesz's lemma.
\begin{proof}
\leanok
The development is dual to the Kolmogorov one under the
domain~$\leftrightarrow$~codomain swap: where the Kolmogorov numbers push
$S$ down a quotient projection $\pi_V : Y \to Y/V$ for a small
$V \subseteq Y$, the Gelfand numbers restrict $S$ along the inclusion
$\iota_M : M \hookrightarrow X$ for a subspace $M \subseteq X$ of small
codimension. With $M = X$ one gets $c_0(S) = \|S\|$; (S2) is the triangle
inequality applied to $(S+T) \circ \iota_M$. For the ideal property (S3),
following Pietsch (\emph{Eigenvalues and $s$-numbers}, \S 2.4), take
admissible $M$ for $S$ and pass to $M' := A^{-1}(M)$: it is closed, its
codimension is at most that of $M$ by the first isomorphism theorem applied
to $\pi_M \circ A$, and $\|BSA(w)\| \le \|B\| \, \|S \circ \iota_M\| \,
\|A\| \, \|w\|$ for $w \in M'$. For (S4) take $M := \ker S$, which is
closed with $\operatorname{codim} M = \operatorname{rank} S \le n$ and
$S \circ \iota_M = 0$. For (S5$'$), the inclusion $\iota_M$ is an isometry,
so the identity restricted to any nontrivial $M$ has norm $1$, and
$\operatorname{codim} M \le n < \dim X$ forces $M \neq \{0\}$. Formalised
in \texttt{Gelfand.lean}.
\end{proof}

\section{Kolmogorov numbers $d_n$}
\label{sec:kol}

\begin{definition}[Kolmogorov number]
\label{def:kolmogorov-number}
\lean{SNumbers.kolmogorovNumber}
\leanok
For $S \in \mathcal{L}(X, Y)$ and $n \in \mathbb{N}_0$,
\[
d_n(S) := \inf_{\substack{V \subseteq Y \\ \dim V \le n}}
            \sup_{\|x\| \le 1} \inf_{v \in V} \|Sx - v\|.
\]
\end{definition}

\begin{theorem}[$d_n$ is a strict $s$-number sequence]
\label{thm:kolmogorov-is-snumber}
\lean{SNumbers.isStrictSNumberSequence_kolmogorovNumber}
\leanok
\uses{def:kolmogorov-number, def:strict-s-number-sequence}
Let $\mathbb{K}$ be a non-trivially normed field and $X, Y$ normed spaces
over $\mathbb{K}$. Then the Kolmogorov numbers $d = (d_n)_n$ form a
\emph{strict} $s$-number sequence: they satisfy (S1)--(S5) and the
strengthening (S5$'$).
\end{theorem}

The (S5$'$) step additionally needs $\mathbb{K}$ \emph{complete} (to close
finite-dimensional subspaces via Riesz's lemma); it requires neither a
densely normed $\mathbb{K}$ nor completeness of $X$.
\begin{proof}
\leanok
The proof works with the operator-norm-on-quotient identity
$d_n(S) = \inf_V \|\pi_V \circ S\|$, where $\pi_V : Y \to Y/V$ is the
canonical projection onto the quotient. (S2) is the triangle inequality
applied to $\pi_V \circ S$; for (S3), an admissible $V$ for $S$ transports
to the admissible subspace $B(V)$ for $BSA$, with the deviation bounded
per point; for (S4) take $V = \operatorname{range} S$. For (S5$'$), given a
subspace $V$ of dimension $\le n < \dim X$, its closure is proper and
Riesz's lemma produces unit vectors almost at distance $1$ from $V$, so
$\|\pi_V\| = 1$ and $d_n(\mathrm{id}_X) = 1$. Formalised in
\texttt{Kolmogorov.lean}.
\end{proof}

\subsection{Alternative: Pietsch's lifting identity}
\label{subsec:kol-lifting}

A second formalization of the Kolmogorov numbers, restricted to the
Banach-space setting, lives in \texttt{KolmogorovLifting.lean}. It uses
Pietsch's identity
\[
  d_n(S) = a_n(S \circ Q_X),
\]
where $a_n$ is the approximation number and
$Q_X : \ell^1(B_X) \twoheadrightarrow X$ is the canonical summation
surjection from the $\ell^1$ space indexed by the closed unit ball
$B_X$, given by
\[
  Q_X(\alpha) := \sum_{x \in B_X} \alpha(x) \cdot x.
\]

\begin{theorem}[$d_n$ via Pietsch's lifting identity]
\label{thm:kolmogorov-lifting}
\lean{SNumbers.Lifting.kolmogorovNumber_strict,
      SNumbers.Lifting.kolmogorovNumber_comp_comp_le,
      SNumbers.Lifting.kolmogorovNumber_zero_eq_norm}
\leanok
\uses{def:kolmogorov-number, def:approximation-number}
For every Banach space $X$ and every $S \in \mathcal{L}(X, Y)$,
$d_n(S) = a_n(S \circ Q_X)$, and the Kolmogorov-number axioms
(S1)--(S5) and (S5$'$) follow from the corresponding axioms for the
approximation numbers.
\end{theorem}
\begin{proof}
\leanok
\uses{thm:approx-is-snumber}
In this variant $d_n(S)$ is \emph{defined} as $a_n(S \circ Q_X)$, so each
axiom reduces to the corresponding fact for the approximation numbers
(Theorem~\ref{thm:approx-is-snumber}). Only the ideal property (S3) needs
an extra construction: it factors as
$(B \circ S \circ A) \circ Q_W = B \circ (S \circ Q_X) \circ T$
through a lift $T : \ell^1(B_W) \to \ell^1(B_X)$ that sends the basis
vector $\delta_w$ to $c \cdot \delta_{c^{-1} A w}$ for a scalar $c$ with
$\|A\| < \|c\|$, so that $\|T\| \le \|c\|$; letting $\|c\| \searrow \|A\|$,
using that the norm values of $\mathbb{K}$ are dense, yields the bound
$\|B\| \, d_n(S) \, \|A\|$. The identification of this $d_n$ with the
canonical Kolmogorov number of the previous section is Pietsch's identity
(Theorem~\ref{thm:pietsch-identity}).
\end{proof}

The lifting development assumes that $\mathbb{K}$ is a densely normed field
(for the scaling above) and that the source spaces $X$, $W$ are complete
(so that $Q_X(\alpha)$ converges as a series in $X$), in addition to
completeness of $\mathbb{K}$ for (S5$'$). On Banach spaces the two
formalisations agree, by Pietsch's identity below.

\begin{theorem}[Pietsch's identity]
\label{thm:pietsch-identity}
\lean{SNumbers.kolmogorovNumber_eq_approx}
\leanok
\uses{def:kolmogorov-number, def:approximation-number}
For every Banach space $X$ and every $S \in \mathcal{L}(X, Y)$, the
canonical Kolmogorov number equals the approximation number of the lift:
$d_n(S) = a_n(S \circ Q_X)$.
\end{theorem}
\begin{proof}
\leanok
\uses{thm:kolmogorov-lifting}
The inequality $d_n(S) \le a_n(S \circ Q_X)$: for a rank-$\le n$ approximant
$L$ of $S \circ Q_X$, take $V := \operatorname{range} L$; since every unit
vector $x$ is $Q_X(\delta_x)$ and $\pi_V \circ L = 0$, one gets
$\|\pi_V \circ S\| \le \|S \circ Q_X - L\|$. The reverse inequality: for an
admissible $V$ and $\varepsilon > 0$, pick $w_x \in V$ with
$\|S x - w_x\| < \|\pi_V \circ S\| + \varepsilon$ and lift to
$L(\alpha) := \sum_x \alpha(x)\, w_x$ (a rank-$\le n$ operator into the
finite-dimensional $V$), so that $\|S \circ Q_X - L\| \le \|\pi_V \circ S\| +
\varepsilon$.
\end{proof}

\section{Hilbert numbers $h_n$}
\label{sec:hil}

For the Hilbert numbers we work over $\mathbb{R}$ and use
$L_2 := \ell_2(\mathbb{N}; \mathbb{R})$.

\begin{definition}[Hilbert number]
\label{def:hilbert-number}
\lean{SNumbers.hilbertNumber}
\leanok
For $S \in \mathcal{L}(X, Y)$ and $n \in \mathbb{N}_0$,
\[
h_n(S) := \sup \biggl\{ \frac{a_n(B \, S \, A)}{\|B\| \, \|A\|} :
            A : L_2 \to X, \ B : Y \to L_2, \ A \neq 0, \ B \neq 0
          \biggr\}.
\]
\end{definition}

\begin{proposition}[$h_n$ is an $s$-number sequence]
\label{prop:hilbert-is-snumber}
\lean{SNumbers.isSNumberSequence_hilbertNumber}
\leanok
\uses{def:hilbert-number, def:approximation-number, def:s-number-sequence}
Let $X, Y$ be Banach spaces. Then the Hilbert numbers $h = (h_n)_n$ form an
$s$-number sequence: they satisfy the axioms (S1)--(S5). The sequence $h$ is
not strict: (S5$'$) fails in general.
\end{proposition}
\begin{proof}
\leanok
Each axiom is obtained by pushing the corresponding property of the
approximation numbers $a_n$ through the supremum defining $h_n$:
nonnegativity, antitonicity, subadditivity (S2), the ideal property (S3)
and the rank axiom (S4) transfer directly. For the normalisations,
$h_0(S) = \|S\|$ combines the upper bound $h_n(S) \le \|S\|$ with a
rank-one factorisation of arbitrary unit vectors through $L_2$, and
$h_n(\mathrm{id}_{\ell_2^{n+1}}) = 1$ (S5) tests the supremum with the
canonical embedding and projection between $\ell_2^{n+1}$ and $L_2$.
Formalised in \texttt{Hilbert.lean}.
\end{proof}

\chapter{Pietsch's sandwich theorem}
\label{ch:sandwich}

\begin{theorem}[Hilbert-space coincidence]
\label{thm:all-eq-on-hilbert}
\lean{SNumbers.allSNumbers_eq_on_HilbertSpace}
\leanok
\uses{def:s-number-sequence, def:approximation-number, thm:scalar-factorisation}
On a Hilbert space, every $s$-number sequence agrees with the
approximation numbers: $s_n(S) = a_n(S)$ for every continuous linear map
$S$ between Hilbert spaces.
\end{theorem}
\begin{proof}
\leanok
The upper bound is Theorem~\ref{thm:sandwich}. For the lower bound
$a_n(S) \le s_n(S)$: given $c < a_n(S)$, the scalar factorisation
(Theorem~\ref{thm:scalar-factorisation}) gives contractions with
$B \circ S \circ A = c \cdot \mathrm{id}_{\ell_2^{n+1}}$; chaining (S3) with
the (S5) normalisation $s_n(\mathrm{id}_{\ell_2^{n+1}}) = 1$ and homogeneity
yields $c \le s_n(S)$. Letting $c \to a_n(S)$ gives $a_n(S) \le s_n(S)$.
\end{proof}

\begin{theorem}[Pietsch \emph{sandwich}]
\label{thm:sandwich}
\lean{SNumbers.hilbertNumber_le_sn_le_approximationNumber}
\leanok
\uses{thm:approx-is-snumber, prop:hilbert-is-snumber, thm:all-eq-on-hilbert}
For every $s$-number sequence $s$ and every operator $S$,
\[
h_n(S) \;\le\; s_n(S) \;\le\; a_n(S).
\]
\end{theorem}
\begin{proof}
\leanok
For the upper bound $s_n(S) \le a_n(S)$, let $A$ be any operator with
$\operatorname{rank}(A) \le n$; then axiom (S2) yields
$s_n(S) \le s_n(A) + \|S - A\| = \|S - A\|$ by (S4), and taking the
infimum over $A$ finishes. For the lower bound $h_n(S) \le s_n(S)$: each
ratio $a_n(B S A)/(\|B\|\|A\|)$ defining $h_n(S)$ has $B S A$ an operator
between Hilbert spaces, so the coincidence theorem
(Theorem~\ref{thm:all-eq-on-hilbert}) gives
$a_n(B S A) = s_n(B S A) \le \|B\| \, s_n(S) \, \|A\|$ by (S3); take the
supremum.
\end{proof}

\chapter{Basic auxiliary results}
\label{ch:basic-results}

The companion library \texttt{BasicResults} (sibling of \texttt{SNumbers})
collects generic functional-analysis results that are independent of the
$s$-numbers framework: Auerbach's lemma (this chapter), the singular value
decomposition of compact operators (Chapter~\ref{ch:svd}), determinant
identities and the bordered-determinant formula, John's ellipsoid with the
Kadets--Snobar and Garling--Gordon projection theorems
(Section~\ref{sec:john-ellipsoid}), the little Grothendieck bounds
(Section~\ref{sec:l1-linf-grothendieck}), and the spectral projection
underlying the Hilbert-space uniqueness theorem.

\begin{theorem}[Auerbach's lemma]
\label{thm:auerbach}
\lean{exists_isAuerbachBasis}
\leanok
Every finite-dimensional normed space over $\mathbb{R}$ of positive
dimension admits a basis $(e_i)$ with $\|e_i\| = 1$ whose dual coordinate
functionals also satisfy $\|e_i^*\| = 1$.
\end{theorem}
\begin{proof}
\leanok
The proof is fully formalized in \texttt{Auerbach.lean}.
The strategy maximises $|\det|$ on a product of unit balls, then reads off
the basis and dual functionals from the maximiser.
\end{proof}

\chapter{Singular value decomposition}
\label{ch:svd}

The singular value decomposition (Schmidt representation) of a compact
operator between Hilbert spaces is formalised in \texttt{BasicResults.SVD},
via the \emph{singular-value iteration} (repeatedly peeling off the top
singular pair). The auxiliary class of \emph{approximable} operators (those
whose approximation numbers $a_n$ tend to zero) and its identification with
the compact operators on Hilbert spaces live in the separate
\texttt{AddOns} library; this split avoids relying on any
approximation-property hypothesis on the underlying space.

\section{Approximable operators}
\label{sec:svd-approximable}

\begin{definition}[Approximable operator]
\label{def:approximable}
\lean{SVD.IsApproximable}
\leanok
\uses{def:approximation-number}
$S \in \mathcal{L}(X, Y)$ is \emph{approximable} if
$a_n(S) \xrightarrow{n \to \infty} 0$, equivalently if $S$ is the
operator-norm limit of a sequence of finite-rank operators.
\end{definition}

\begin{proposition}[Closure properties of approximable operators]
\label{prop:approximable-closure}
\lean{SVD.isApproximable_iff_existsLimit,
      SVD.isApproximable_zero,
      SVD.IsApproximable.add,
      SVD.IsApproximable.smul,
      SVD.IsApproximable.comp_comp,
      SVD.IsApproximable.of_rank_le,
      SVD.isClosed_isApproximable}
\leanok
\uses{def:approximable, def:approximation-number}
The approximable operators form a closed two-sided operator ideal: a
subspace of $\mathcal{L}(X, Y)$, closed under pre/post-composition with
bounded operators, closed in operator-norm topology, and containing all
finite-rank operators.
\end{proposition}
\begin{proof}
\leanok
The equivalence with being a norm-limit of finite-rank operators follows by
choosing near-optimal rank-$\le n$ approximants $L_n$ with
$\|S - L_n\| < a_n(S) + \tfrac{1}{n+1}$, and conversely by squeezing
$a_n(S) \le \|S - L_n\|$. Stability under addition combines rank-$\le
\lfloor n/2 \rfloor$ and rank-$\le \lceil n/2 \rceil$ approximants of the
two summands, using that ranks are subadditive; stability under
composition and closedness in operator norm are the axioms (S3) and (S2)
of $a_n$ passed through the limit. Formalised in
\texttt{Approximable.lean}.
\end{proof}

\begin{theorem}[Approximable $\Rightarrow$ compact]
\label{thm:approximable-implies-compact}
\lean{SVD.IsApproximable.isCompactOperator,
      SVD.isCompactOperator_of_rank_le}
\leanok
\uses{def:approximable, prop:approximable-closure}
Every approximable operator $S : X \to Y$ from a normed space $X$ to a
Banach space $Y$ over a locally compact field $\mathbb{K}$ is compact. In
particular every operator of finite rank is compact, since finite-rank
operators are approximable (Proposition~\ref{prop:approximable-closure}).
\end{theorem}
\begin{proof}
\leanok
Each finite-rank approximant $L_n$ corestricts to its range, a
finite-dimensional subspace of $Y$, which is locally compact because
$\mathbb{K}$ is; an operator into a locally compact space is compact, and
post-composition with the inclusion preserves compactness. Since
compactness passes to operator-norm limits, $S = \lim_n L_n$ is compact.
\end{proof}

\section{Schmidt representation}
\label{sec:svd-schmidt}

For the rest of this chapter, $H_1, H_2$ are Hilbert spaces over
$\mathbb{K} \in \{\mathbb{R}, \mathbb{C}\}$.

\begin{theorem}[Norm attained as a singular value]
\label{thm:norm-singular-value}
\lean{SVD.IsCompactOperator.norm_isSingularValue}
\leanok
Every compact $S \in \mathcal{L}(H_1, H_2)$ attains its operator norm as a
singular value: there are unit vectors $u, v$ with $S u = \|S\| \, v$ and
$S^* v = \|S\| \, u$.
\end{theorem}
\begin{proof}
\leanok
Take a maximising sequence $x_n$ of unit vectors with $\|S x_n\| \to \|S\|$;
by compactness extract a subsequence with $S x_{n_k} \to y$, and set
$v := \|S\|^{-1} y$, $u' := S^* v$. Cauchy--Schwarz on
$\langle u', x_{n_k}\rangle = \langle v, S x_{n_k}\rangle \to \|S\|$ gives
$\|u'\| = \|S\|$; equality in Cauchy--Schwarz forces $x_{n_k} \to u$, so by
continuity $S u = y = \|S\| v$.
\end{proof}

\begin{theorem}[Singular value decomposition / Schmidt representation]
\label{thm:schmidt-representation}
\lean{SVD.IsCompactOperator.SVD}
\leanok
\uses{def:approximation-number, thm:norm-singular-value}
For every compact $S \in \mathcal{L}(H_1, H_2)$ there exist
orthonormal sequences $(u_k) \subseteq H_1$, $(v_k) \subseteq H_2$ such
that
\[
  S x \;=\; \sum_{k} a_k(S) \, \langle u_k, x \rangle \, v_k
  \qquad \text{for every } x \in H_1.
\]
The singular values are exactly the approximation numbers $a_k(S)$.
\end{theorem}
\begin{proof}
\leanok
\uses{thm:norm-singular-value}
By iterating Theorem~\ref{thm:norm-singular-value}: peel off the top
singular pair $(\sigma_0, u_0, v_0)$ of $S$, replace $S$ by
$S - \sigma_0 \langle u_0, \cdot\rangle v_0$ (which is again compact), and
recurse. The variational characterisation of each $u_k$ forces the $u_k$
(and dually the $v_k$) to be orthonormal; $\sigma_k \to 0$ follows from
compactness; and the partial sums converge to $S$ in operator norm because
the $n$-th residual has norm $\sigma_n$. Eckart--Young
(Theorem~\ref{thm:eckart-young}) identifies $\sigma_k = a_k(S)$.
\end{proof}

\begin{theorem}[Eckart--Young: optimal low-rank approximation]
\label{thm:eckart-young}
\lean{SVD.IsCompactOperator.truncation_residual_eq_approxNumber}
\leanok
\uses{def:approximation-number, thm:schmidt-representation}
For every approximable $S \in \mathcal{L}(H_1, H_2)$ and every
$n \in \mathbb{N}_0$ there exists $L \in \mathcal{L}(H_1, H_2)$ with
$\operatorname{rank}(L) \le n$ and $\|S - L\| = a_n(S)$.
Concretely, $L = \sum_{k < n} a_k(S) \langle u_k, \cdot \rangle v_k$ is
the truncated SVD.
\end{theorem}
\begin{proof}
\leanok
\uses{thm:schmidt-representation}
Take the truncated SVD $L = \sum_{k < n} a_k(S)\langle u_k, \cdot\rangle v_k$
from Theorem~\ref{thm:schmidt-representation}, which has rank $\le n$. The
residual $S - L = \sum_{k \ge n} a_k(S)\langle u_k, \cdot\rangle v_k$ acts on the
orthonormal $(u_k)_{k \ge n}$ with weights $a_k(S) \le a_n(S)$, so its operator
norm is exactly $a_n(S)$; no rank-$\le n$ operator does better, since $a_n(S)$
is by definition the infimum of $\|S - L'\|$ over such $L'$.
\end{proof}

\begin{theorem}[Diagonal factorisation through $\ell_2^{n+1}$]
\label{thm:schmidt-diagonal}
\lean{SVD.IsCompactOperator.diagonalFactorisation}
\leanok
\uses{def:approximation-number, thm:schmidt-representation}
For every compact $S \in \mathcal{L}(H_1, H_2)$ and every
$n \in \mathbb{N}_0$ there exist contractions
$A \in \mathcal{L}(\ell_2^{n+1}, H_1)$ and
$B \in \mathcal{L}(H_2, \ell_2^{n+1})$ such that, on every basis vector
$e_k$ of $\ell_2^{n+1}$,
\[
  (B \circ S \circ A)(e_k) \;=\; a_k(S) \, e_k,
  \qquad 0 \le k \le n.
\]
Equivalently, $B \circ S \circ A$ is the diagonal operator
$\operatorname{diag}(a_0(S), \dots, a_n(S))$ on $\ell_2^{n+1}$.
\end{theorem}
\begin{proof}
\leanok
\uses{thm:schmidt-representation}
Using the top $n+1$ singular pairs of Theorem~\ref{thm:schmidt-representation},
let $A \in \mathcal{L}(\ell_2^{n+1}, H_1)$ send the basis vector $e_k$ to the
right singular vector $u_k$, and let $B \in \mathcal{L}(H_2, \ell_2^{n+1})$ send
$y$ to $\sum_{k \le n} \langle v_k, y\rangle e_k$, i.e.\ project onto the span of
the left singular vectors. Both are contractions because $(u_k)$ and $(v_k)$ are
orthonormal. Then $S u_k = a_k(S)\, v_k$ gives
$(B \circ S \circ A) e_k = a_k(S)\, e_k$ for $0 \le k \le n$.
\end{proof}

This factorisation needs only the top $n+1$ singular pairs --- a finite
truncation, a sibling of the full SVD rather than a corollary --- but,
pinning the \emph{exact} values $a_k$, it requires $S$ compact.

\begin{theorem}[Scalar factorisation]
\label{thm:scalar-factorisation}
\lean{SVD.exists_scalar_factorisation}
\leanok
\uses{def:approximation-number}
For \emph{any} bounded $S \in \mathcal{L}(H_1, H_2)$ between Hilbert spaces
and any real $c$ with $0 \le c < a_n(S)$, there are contractions
$A \in \mathcal{L}(\ell_2^{n+1}, H_1)$, $B \in \mathcal{L}(H_2, \ell_2^{n+1})$
with $B \circ S \circ A = c \cdot \mathrm{id}_{\ell_2^{n+1}}$.
\end{theorem}
\begin{proof}
\leanok
\uses{def:approximation-number}
Since $c < a_n(S)$, no rank-$\le n$ operator approximates $S$ to within $c$, so
there is an $(n+1)$-dimensional subspace $M \subseteq H_1$ on which
$\|S x\| \ge c\,\|x\|$. Let $A$ embed $\ell_2^{n+1}$ isometrically onto $M$; then
$S \circ A$ is bounded below by $c$, so $c\,(S \circ A)^{-1}$ is a contraction on
its range, and composing it with the orthogonal projection onto that range gives
a contraction $B$ with $B \circ S \circ A = c \cdot \mathrm{id}_{\ell_2^{n+1}}$.
No compactness is used.
\end{proof}

Unlike the diagonal factorisation this holds for \emph{arbitrary} bounded
$S$ (no compactness): the strict inequality $c < a_n(S)$ needs only an
$(n+1)$-dimensional subspace on which $\|S x\| \ge c\|x\|$, which exists for
every bounded operator.

\section{Leading singular values of a product of Hilbert--Schmidt operators}
\label{sec:svd-hs-product}

The Schmidt representation also yields the estimate behind the classical
statement that the nuclear norm of a product is at most the product of the
Hilbert--Schmidt norms of its factors (Pietsch, \emph{Eigenvalues and
$s$-numbers}, 2.11.23). We need it only in the following quantitative form, and
we never define the Hilbert--Schmidt or nuclear norms: the two Hilbert--Schmidt
bounds enter as hypotheses.

\begin{lemma}[Bessel plus a row bound]
\label{lem:bessel-rows}
\lean{SVD.sum_norm_sq_apply_le_of_rows}
\leanok
Let $S \in \mathcal{L}(H_1, H_2)$ be described by \emph{rows} $w_j \in H_1$, in
the sense that $\|S x\|^2 = \sum_j |\langle w_j, x\rangle|^2$ for all $x$, and
suppose $\sum_{j \in J} \|w_j\|^2 \le c$ for every finite $J$. Then
$\sum_{k \in s} \|S u_k\|^2 \le c$ for every finite $s$ and every family
$(u_k)$ of pairwise orthogonal vectors of norm $1$ or $0$.
\end{lemma}
\begin{proof}
\leanok
Interchange the two summations and apply Bessel's inequality
$\sum_k |\langle u_k, w_j\rangle|^2 \le \|w_j\|^2$ for each fixed $j$, then sum
over $j$ and use the row bound.
\end{proof}

\begin{theorem}[Leading singular values of a product]
\label{thm:hs-product-bound}
\lean{SVD.mul_approximationNumber_le_of_factorization}
\leanok
\uses{thm:schmidt-representation, thm:eckart-young, def:approximation-number}
Let $T = T_2 T_1$ be compact, with $T_1 \in \mathcal{L}(H_1, H_2)$ and
$T_2 \in \mathcal{L}(H_2, H_3)$, and let $a, b \ge 0$ be such that
\[
  \sum_{k \in s} \|T_1 u_k\|^2 \le a^2
  \qquad\text{and}\qquad
  \sum_{k \in s} \|T_2^{*} v_k\|^2 \le b^2
\]
for all finite $s$ and all orthonormal(-or-zero) families $(u_k)$ in $H_1$ and
$(v_k)$ in $H_3$. Then
\[
  (n+1)\, a_n(T) \;\le\; \sum_{k=0}^{n} \sigma_k(T) \;\le\; b\,a
  \qquad (n \in \mathbb{N}_0).
\]
\end{theorem}
\begin{proof}
\leanok
\uses{thm:schmidt-representation, thm:eckart-young}
Take a Schmidt representation of $T$ with singular values $\sigma_k$ and
singular vectors $u_k, v_k$; by Eckart--Young $\sigma_n = a_n(T)$, and since
$\sigma$ is antitone, $(n+1)\sigma_n \le \sum_{k \le n} \sigma_k$. For each $k$
with $\sigma_k \ne 0$ the vector $v_k$ is a unit vector and $T u_k = \sigma_k v_k$,
whence
\[
  \sigma_k = \operatorname{Re}\langle v_k, T u_k\rangle
           = \operatorname{Re}\langle T_2^{*} v_k, T_1 u_k\rangle
           \le \|T_2^{*} v_k\|\,\|T_1 u_k\| .
\]
Summing over $k \le n$ and applying Cauchy--Schwarz together with the two
hypotheses gives $\sum_{k \le n}\sigma_k \le b\,a$.
\end{proof}

\section{Compact $\Leftrightarrow$ approximable on Hilbert spaces}
\label{sec:svd-compact-equiv}

\begin{theorem}[Compact $\Rightarrow$ approximable on Hilbert spaces]
\label{thm:compact-implies-approximable}
\lean{SVD.IsCompactOperator.isApproximable}
\leanok
\uses{def:approximable, thm:schmidt-representation}
Every compact $S \in \mathcal{L}(H_1, H_2)$ between Hilbert spaces is
approximable.
\end{theorem}
\begin{proof}
\leanok
The singular value decomposition gives singular values $\sigma_n \to 0$ which, by
Eckart--Young, equal the approximation numbers $a_n(S)$; hence $a_n(S) \to 0$.
Equivalently, the SVD truncations are finite-rank operators converging to $S$ in
operator norm.
\end{proof}

\begin{theorem}[Compact $\Leftrightarrow$ approximable on Hilbert spaces]
\label{thm:compact-iff-approximable}
\lean{SVD.isApproximable_iff_isCompactOperator}
\leanok
\uses{thm:approximable-implies-compact, thm:compact-implies-approximable}
On Hilbert spaces, $S$ is approximable iff $S$ is compact.
\end{theorem}
\begin{proof}
\leanok
\uses{thm:approximable-implies-compact, thm:compact-implies-approximable}
Combine the two implications: approximable $\Rightarrow$ compact is
Theorem~\ref{thm:approximable-implies-compact}, and compact $\Rightarrow$
approximable is Theorem~\ref{thm:compact-implies-approximable}.
\end{proof}

\section{All $s$-numbers equal the singular values in finite dimension}
\label{sec:svd-findim}

Mathlib defines the \emph{singular values} of a linear map $S$ between
finite-dimensional Hilbert spaces as
$\sigma_n(S) := \sqrt{\lambda_n(S^* S)}$, where $\lambda_0 \ge \lambda_1
\ge \dots$ are the eigenvalues of $S^* S$ in non-increasing order. The
following theorem identifies them with \emph{every} $s$-number sequence at
once.

\begin{theorem}[All $s$-numbers equal Mathlib's singular values]
\label{thm:sn-eq-singular-values}
\lean{SNumbers.sn_eq_singularValues_of_finiteDimensional,
      SNumbers.project_singularValues_eq}
\leanok
\uses{def:s-number-sequence, thm:all-eq-on-hilbert, thm:schmidt-representation,
  thm:eckart-young}
Let $H_1, H_2$ be finite-dimensional Hilbert spaces over
$\mathbb{K} \in \{\mathbb{R}, \mathbb{C}\}$. For every $s$-number sequence
$s$, every $S \in \mathcal{L}(H_1, H_2)$ and every $n \in \mathbb{N}_0$,
we have
\[
  s_n(S) \;=\; \sigma_n(S),
\]
where $\sigma_n(S)$ are Mathlib's eigenvalue-defined singular values.
\end{theorem}
\begin{proof}
\leanok
\uses{thm:all-eq-on-hilbert, thm:schmidt-representation, thm:eckart-young}
The Hilbert-space coincidence (Theorem~\ref{thm:all-eq-on-hilbert}) reduces
$s_n(S)$ to $a_n(S)$, so it suffices to identify $a_n(S)$ with
$\sigma_n(S)$. The operator $S$ is compact (finite-dimensional domain), so
the Schmidt representation (Theorem~\ref{thm:schmidt-representation})
applies and its singular values equal the approximation numbers by
Eckart--Young. On the other hand, the eigenvalues of $S^* S$ are exactly
the squares of the Schmidt singular values (with the singular vectors as
an eigenbasis), so Mathlib's $\sigma_n(S) = \sqrt{\lambda_n(S^* S)}$
coincides with the $n$-th Schmidt singular value; for
$n \ge \dim H_1$ both sides vanish. Formalised in
\texttt{SingularValuesFinDim.lean}.
\end{proof}

\chapter{Inequalities between \texorpdfstring{$s$}{s}-numbers}
\label{ch:inequalities}

Beyond the qualitative sandwich $h_n(S) \le s_n(S) \le a_n(S)$
(Theorem~\ref{thm:sandwich}), one can compare the individual example
sequences \emph{quantitatively}. This chapter formalises two such
comparisons for operators between Banach spaces over $\mathbb{K}$:

\begin{itemize}
  \item the \emph{forward} bound $a_n(S) \le (1+\sqrt{n})\min(c_n(S), d_n(S))$
        --- the largest $s$-number exceeds the Gelfand and Kolmogorov numbers by
        at most a factor $1+\sqrt{n}$ (classical; see Pietsch, \emph{s-Numbers
        of operators in Banach spaces}, Studia Math.\ \textbf{51} (1974), and
        \emph{Eigenvalues and $s$-Numbers}, Cambridge, 1987, §2.10);
  \item the \emph{reverse} bound, the \emph{maximal difference theorem}
        $a_n(S) \le e\,(n+1)\, h_n(S)$ --- the maximal possible difference
        between the largest $s$-numbers, the approximation numbers, and the
        smallest ones, the Hilbert numbers, is a factor linear in $n$
        (Ullrich, \emph{On bounds between all $s$-numbers and widths of
        convex sets}, preprint, 2026). Its consequences for two arbitrary
        $s$-number sequences (Carl--Pietsch) and for the Gelfand, Kolmogorov and
        Bernstein numbers (Mityagin--Henkin) are collected in
        Corollary~\ref{cor:gelfand-kolmogorov}, both up to the constant $e$.
\end{itemize}

Everything in this chapter is proved in \texttt{Inequalities.lean}
and \texttt{MaxDifference.lean}. The two classical projection theorems
that drive the forward bound (Garling--Gordon and Kadets--Snobar) are proved,
for $\mathbb{K} \in \{\mathbb{R}, \mathbb{C}\}$ at once, by the
John's-ellipsoid development of Section~\ref{sec:john-ellipsoid}, whose
analytic core is the John decomposition of identity
(Theorem~\ref{thm:john-decomposition}).

\section{John's ellipsoid and the projection theorems}
\label{sec:john-ellipsoid}

The sharp constant $\sqrt{n}$ in both classical projection theorems below rests
on \emph{John's ellipsoid theorem}, which is absent from Mathlib. It is developed
in \texttt{John.lean} over $\mathbb{K} \in \{\mathbb{R},
\mathbb{C}\}$ at once, modelling an inscribed ellipsoid as the image $T(B_2)$
of the Euclidean unit ball under a linear map and maximising its volume
$|\det T|$. The general-purpose ingredients of the analytic core
(compactness of convex hulls of compact sets in finite dimension,
Hahn--Banach dominated by a seminorm, trace duality, and a Weierstrass-type
product bound) are collected separately, as candidates for Mathlib.

\begin{theorem}[Maximal-volume inscribed ellipsoid]
\label{thm:john-maxvolume}
\lean{John.exists_maxVolume}
\leanok
For a body seminorm $p$ on $\mathbb{K}^k$ equivalent to the Euclidean norm, among
the \emph{feasible} operators (those $T$ with $p(Tu) \le \|u\|$ for all $u$) there
is one, $T_0$, of maximal $|\det|$, and it is invertible ($\det T_0 \ne 0$).
\end{theorem}
\begin{proof}
\leanok
The feasible set is closed (each $T \mapsto p(Tu)$ is continuous) and bounded
(the lower norm bound forces $\|T\| \le 1/c$), hence compact; $|\det|$ is
continuous, so it attains a maximum. A small multiple of the identity is feasible
and has nonzero determinant, so the maximiser does too.
\end{proof}

\begin{theorem}[John decomposition of identity]
\label{thm:john-decomposition}
\lean{John.john_decomposition}
\leanok
In John position --- the identity is a maximal-volume feasible operator for the
body seminorm $q$ --- the identity is a positive combination of the rank-one
projections onto contact points:
\[
  \sum_i c_i \, \langle u_i, x\rangle \, u_i = x \quad (x \in \mathbb{K}^k),
  \qquad u_i \in \operatorname{Contact}(q),\ c_i \ge 0,\ \sum_i c_i = k,
\]
where a contact point is a unit vector $u$ with $\operatorname{Re}\langle x, u\rangle
\le q(x)$ for all $x$.
\end{theorem}
\begin{proof}
\leanok
The classical core of John's theorem, from first-order optimality of the
maximal-volume ellipsoid. If $k^{-1}\mathrm{id}$ lay outside the convex hull of the
contact rank-ones --- a compact convex set of operators, whose compactness uses
that convex hulls of compact sets are compact in finite dimension --- then
geometric Hahn--Banach separation, followed by trace duality, would produce a
self-adjoint trace-zero direction $H$ with $\operatorname{Re}\langle u, Hu\rangle
< 0$ at every contact point. The perturbation $(1-\rho)^{-1}(\mathrm{id} + tH)$
then stays feasible for small $t>0$ (the feasibility slack grows linearly in $t$,
using that every touching point is a contact point via a seminorm-dominated
Hahn--Banach argument), while its determinant loses only $O(t^2)$ because
$\operatorname{tr} H = 0$; so $|\det| > 1$, contradicting maximality. Hence
$k^{-1}\mathrm{id}$ lies in the hull, and Carathéodory yields the finite
combination.
\end{proof}

\begin{theorem}[Kadets--Snobar projection theorem]
\label{thm:kadets-snobar}
\lean{SNumbers.exists_projection_range_eq_of_rank_le}
\leanok
\uses{thm:john-decomposition}
Let $Y$ be a normed $\mathbb{K}$-space and $V \subseteq Y$ a subspace of dimension
$\le n$. Then there is a bounded projection $P : Y \to Y$ ($P \circ P = P$) with
$\operatorname{range} P = V$ and $\|P\| \le \sqrt{n}$.
\end{theorem}
\begin{proof}
\leanok
\uses{thm:john-decomposition}
A classical result (Kadets--Snobar 1971; Pietsch 1.5.5). Put $V$ in John position
via the maximal-volume ellipsoid: the decomposition of identity
(Theorem~\ref{thm:john-decomposition}) gives contact points $u_i$, unit vectors
$v_i = M u_i \in V$, and functionals $\varphi_i = \langle u_i, M^{-1}\cdot\rangle$
of norm $\le 1$, which Hahn--Banach extends to $g_i \in Y^*$ without increasing
the norm. Then $P = \sum_i c_i\, g_i \otimes v_i$ is the identity on $V$, hence a
projection onto $V$, and weighted Cauchy--Schwarz with $\sum_i c_i = \dim V$ gives
$\|P\| \le \sqrt{\dim V} \le \sqrt{n}$. Formalised in \texttt{John.exists\_projection}.
\end{proof}

\begin{theorem}[Garling--Gordon projection theorem, $\varepsilon$-form]
\label{thm:garling-gordon}
\lean{SNumbers.exists_projection_ker_eq_of_codim_le}
\leanok
\uses{thm:john-decomposition}
Let $X$ be a normed $\mathbb{K}$-space and $M \subseteq X$ a closed subspace of
codimension $\le n$. Then for every $\varepsilon > 0$ there is a bounded
projection $P : X \to X$ ($P \circ P = P$) with $\ker P = M$ and
$\|P\| \le \sqrt{n} + \varepsilon$.
\end{theorem}
\begin{proof}
\leanok
\uses{thm:john-decomposition}
The dual of Kadets--Snobar (Garling--Gordon 1971; Pietsch 1.7.17). Run the John
development on the finite-dimensional dual $(X/M)^*$: the contact points are
represented by vectors $w_i \in X/M$ with $\|w_i\| \le 1$ (obtained by solving
the single-dual equation $\Phi^{\mathrm{flip}} w_i = \operatorname{toDual} u_i$,
avoiding the topological double dual), each lifted to $x_i \in X$ with
$\|x_i\| < 1 + \varepsilon'$ (the quotient norm is an infimum --- the sole source
of $\varepsilon$). Then $P y = \sum_i c_i\,(\Phi u_i)(\pi y)\, x_i$ has kernel $M$
and $\|P\| \le \sqrt{\operatorname{codim} M} + \varepsilon \le \sqrt{n} +
\varepsilon$. Formalised in \texttt{John.exists\_projection\_ker}. The exact bound
$\sqrt{n}$ (removing $\varepsilon$) needs a weak-$*$ limit argument; it is not
needed here, since the applications recover the sharp constant by letting
$\varepsilon \to 0$.
\end{proof}

\section{Approximation versus Gelfand and Kolmogorov numbers}
\label{sec:ineq-forward}

\begin{theorem}[$a_n \le (1+\sqrt n)\,c_n$]
\label{thm:approx-le-gelfand}
\lean{SNumbers.approximationNumber_le_sqrt_mul_gelfandNumber}
\leanok
\uses{def:approximation-number, def:gelfand-number, thm:garling-gordon}
For $S \in \mathcal{L}(X, Y)$ between Banach spaces,
$a_n(S) \le (1 + \sqrt{n})\,c_n(S)$.
\end{theorem}
\begin{proof}
\leanok
\uses{thm:garling-gordon}
Fix a closed $M \subseteq X$ of codimension $\le n$ with $\|S|_M\|$ near
$c_n(S)$. By Garling--Gordon (Theorem~\ref{thm:garling-gordon}), for each
$\varepsilon>0$ choose a projection $P$ with $\ker P = M$ and
$\|P\| \le \sqrt n + \varepsilon$. Then $L := S \circ P$ has rank
$\le \operatorname{codim} M \le n$, and $S - L = S \circ (\mathrm{id} - P)$ has
range inside $M$ (since $\operatorname{range}(\mathrm{id}-P) = \ker P = M$), so
$\|S - L\| \le \|S|_M\|\,\|\mathrm{id} - P\| \le (1+\sqrt n+\varepsilon)\,\|S|_M\|$.
Letting $\varepsilon \to 0$ gives $a_n(S) \le (1+\sqrt n)\,\|S|_M\|$; taking the
infimum over $M$ gives the claim.
\end{proof}

\begin{theorem}[$a_n \le (1+\sqrt n)\,d_n$]
\label{thm:approx-le-kolmogorov}
\lean{SNumbers.approximationNumber_le_sqrt_mul_kolmogorovNumber}
\leanok
\uses{def:approximation-number, def:kolmogorov-number, thm:kadets-snobar}
For $S \in \mathcal{L}(X, Y)$ between Banach spaces,
$a_n(S) \le (1 + \sqrt{n})\,d_n(S)$.
\end{theorem}
\begin{proof}
\leanok
\uses{thm:kadets-snobar}
Dual to the Gelfand half. Fix $V \subseteq Y$ of dimension $\le n$ with
$\|\pi_V \circ S\|$ near $d_n(S)$; by Kadets--Snobar
(Theorem~\ref{thm:kadets-snobar}) choose a projection $P$ onto $V$ with
$\|P\| \le \sqrt n$. Then $L := P \circ S$ has rank $\le n$, and
$S - L = (\mathrm{id} - P) \circ S$ factors through the quotient $Y / V$
(because $\mathrm{id} - P$ vanishes on $V$), so
$\|S - L\| \le \|\mathrm{id} - P\|\,\|\pi_V \circ S\| \le (1+\sqrt n)\,d_n(S)$
after taking the infimum.
\end{proof}

\begin{theorem}[Combined bound $a_n \le (1+\sqrt n)\min(c_n, d_n)$]
\label{thm:approx-le-min}
\lean{SNumbers.approximationNumber_le_sqrt_mul_min}
\leanok
\uses{thm:approx-le-gelfand, thm:approx-le-kolmogorov}
For $S \in \mathcal{L}(X, Y)$ between Banach spaces,
$a_n(S) \le (1 + \sqrt{n})\,\min\bigl(c_n(S), d_n(S)\bigr)$.
\end{theorem}
\begin{proof}
\leanok
\uses{thm:approx-le-gelfand, thm:approx-le-kolmogorov}
Immediate from the two halves (Theorems~\ref{thm:approx-le-gelfand}
and~\ref{thm:approx-le-kolmogorov}) by case analysis on which of $c_n(S)$,
$d_n(S)$ is the smaller.
\end{proof}

\section{The maximal difference theorem}
\label{sec:ineq-maxdiff}

The maximal difference theorem $a_n \le e\,(n+1)\, h_n$ --- a bound between the
largest and the smallest $s$-number sequence, and hence between any two --- is
driven by the \emph{determinant quantities}
\[
  \Delta_k(S) \;:=\; \sup\bigl\{\, \|\det(B \circ S \circ A)\| \,\bigr\},
\]
where the supremum ranges over all contractions $A : \ell_2^k \to X$ and
$B : Y \to \ell_2^k$. They are formalised, together with the theorems
themselves, in \texttt{MaxDifference.lean} (the bordered-determinant lemma,
a pure matrix fact, in \texttt{Determinant.lean}). The $\Delta_k$ are
squeezed between the two quantities of interest: they do not decay faster than
the approximation numbers allow (the growth lemma below, via the explicit
rank-$k$ approximant $L = SA(BSA)^{-1}BS$), and consecutive ratios are bounded
by the Hilbert numbers (the upper bound below, via contractive compressions).

\begin{definition}[Determinant quantities]
\label{def:det-quantity}
\lean{SNumbers.detNumber}
\leanok
For $S \in \mathcal{L}(X, Y)$ and $k \in \mathbb{N}_0$, $\Delta_k(S)$ is the
supremum above; here $B \circ S \circ A$ is an endomorphism of the
$k$-dimensional Hilbert space $\ell_2^k$, so its determinant is a scalar.
One has $\Delta_0(S) = 1$ (the determinant on the trivial space is $1$) and
$\Delta_k(S) \le \|S\|^k$, via $\prod_i a_i(T) = \|\det T\|$
(Lemma~\ref{lem:prod-approx-eq-det}) and $a_i(T) \le \|T\| \le \|S\|$.
\end{definition}

\begin{lemma}[$\prod_k a_k(T) = \|\det T\|$]
\label{lem:prod-approx-eq-det}
\lean{SNumbers.prod_approximationNumber_eq_norm_det}
\leanok
\uses{def:approximation-number, thm:sn-eq-singular-values}
For an endomorphism $T$ of a finite-dimensional inner-product space,
$\prod_{k < \dim} a_k(T) = \|\det T\|$.
\end{lemma}
\begin{proof}
\leanok
\uses{thm:sn-eq-singular-values}
The approximation numbers coincide with the singular values $\sigma_k(T)$
(Theorem~\ref{thm:sn-eq-singular-values}), and
$\|\det T\| = \prod_k \sigma_k(T)$ is linear algebra:
$\|\det T\|^2 = \det(T^* T) = \prod_k \lambda_k(T^* T) = \prod_k
\sigma_k(T)^2$.
\end{proof}

\begin{lemma}[Bordered determinant]
\label{lem:det-bordered}
\lean{Matrix.det_eq_corner_mul_det_submatrix}
\leanok
Let $M$ be a $(k+1) \times (k+1)$ matrix and $w \in \mathbb{K}^k$ such that,
above the corner, the last column of $M$ is the combination of the first $k$
columns with coefficients $w$, i.e.\ $M_{i,\mathrm{last}} = \sum_j M_{ij} w_j$
for $i < k$. Then
\[
  \det M \;=\;
  \Bigl( M_{\mathrm{last},\mathrm{last}}
    - \sum_j M_{\mathrm{last},j}\, w_j \Bigr)
  \cdot \det M' ,
\]
where $M'$ is the top-left $k \times k$ block of $M$.
No invertibility of the top-left block is required, and the statement holds
over any commutative ring; this is the elementary column-operation form of
the Schur determinant formula.
\end{lemma}
\begin{proof}
\leanok
Subtracting from the last column the combination of the other columns with
coefficients $w$ leaves the determinant unchanged and clears the column above
the corner; Laplace expansion along it gives the claim.
\end{proof}

\begin{lemma}[Growth lemma]
\label{lem:det-growth}
\lean{SNumbers.approximationNumber_mul_detNumber_le_detNumber_succ}
\leanok
\uses{def:det-quantity, lem:det-bordered, def:approximation-number}
For every $S \in \mathcal{L}(X, Y)$ and $k \in \mathbb{N}_0$,
\[
  \frac{k^k}{(k+1)^{k+1}}\; a_k(S)\; \Delta_k(S) \;\le\; \Delta_{k+1}(S).
\]
In particular, if $a_n(S) > 0$ then $\Delta_k(S) > 0$ for all $k \le n+1$, by
induction from $\Delta_0(S) = 1$ and $a_k(S) \ge a_n(S)$.
\end{lemma}
\begin{proof}
\leanok
\uses{lem:det-bordered}
Fix an admissible pair $(A, B)$ for $\Delta_k(S)$ and put $T := BSA$; if
$\det T = 0$ there is nothing to prove. Otherwise $T$ is invertible, so
$L := SA\,T^{-1}BS$ factors through $\ell_2^k$, hence has rank at most $k$,
hence $\|S - L\| \ge a_k(S)$ by the definition of the approximation numbers.
Pick $x \in B_X$ with $\|(S-L)x\|$ almost $\|S - L\|$ and a Hahn--Banach
functional $b \in B_{Y^*}$ norming $(S-L)x$, and border the pair by them:
$A_0(\xi \oplus \tau) := \lambda\, A\xi + \mu\, \tau\, x$ and
$B_0 y := (\lambda\, B y) \oplus (\mu\, b(y))$ with
$\lambda^2 + \mu^2 = 1$. Contractivity of $A_0$ is the two-term
Cauchy--Schwarz inequality $\lambda a + \mu b \le \sqrt{a^2+b^2}$, and
$\|B_0 y\|^2 = \lambda^2\|By\|^2 + \mu^2 |b(y)|^2 \le \|y\|^2$. Above the
corner, the last column of the matrix of $B_0SA_0$ is the combination of the
first $k$ columns with coefficients $(\mu/\lambda)\, \zeta$, where
$\zeta := T^{-1}B(Sx)$, so Lemma~\ref{lem:det-bordered} gives
\[
  \det(B_0 S A_0) \;=\; \lambda^{2k} \mu^2\, b\bigl((S-L)x\bigr) \det T .
\]
The weights $\lambda^2 = k/(k+1)$, $\mu^2 = 1/(k+1)$ maximise
$\lambda^{2k}\mu^2$, whose value is then exactly $k^k/(k+1)^{k+1}$; taking
the supremum over $(A, B)$ finishes. Note that no near-maximiser has to be
selected --- \emph{every} admissible pair is improved.
\end{proof}

\begin{lemma}[Upper bound $\Delta_{n+1} \le h_n \Delta_n$]
\label{lem:det-upper}
\lean{SNumbers.detNumber_succ_le_hilbertNumber_mul}
\leanok
\uses{def:det-quantity, lem:prod-approx-eq-det, thm:schmidt-diagonal, def:hilbert-number}
$\Delta_{n+1}(S) \le h_n(S)\, \Delta_n(S)$.
\end{lemma}
\begin{proof}
\leanok
\uses{lem:prod-approx-eq-det, thm:schmidt-diagonal}
For admissible $(A, B)$ and $T := B S A$ on $\ell_2^{n+1}$,
$\|\det T\| = \prod_{k \le n} a_k(T)$ (Lemma~\ref{lem:prod-approx-eq-det}).
The last factor satisfies $a_n(T) \le \|B\|\|A\|\, h_n(S) \le h_n(S)$. For the
top $n$ factors, the diagonal factorisation
$B_1 T A_1 = \operatorname{diag}(a_0(T), \dots, a_n(T))$ through contractions
(Theorem~\ref{thm:schmidt-diagonal}), compressed to the first $n$ coordinates
by the coordinate embedding/projection, is itself admissible for
$\Delta_n(S)$, whence $\prod_{k < n} a_k(T) \le \Delta_n(S)$. Again every
admissible pair is bounded, so the bound passes to the supremum
$\Delta_{n+1}(S)$.
\end{proof}

\begin{theorem}[Maximal difference theorem]
\label{thm:maximal-difference}
\lean{SNumbers.approximationNumber_le_mul_hilbertNumber,
      SNumbers.approximationNumber_le_e_mul_hilbertNumber}
\leanok
\uses{lem:det-growth, lem:det-upper, def:approximation-number, def:hilbert-number}
For every $S \in \mathcal{L}(X, Y)$ and $n \in \mathbb{N}_0$,
\[
  a_n(S) \;\le\; \frac{(n+1)^{n+1}}{n^n}\, h_n(S)
  \;\le\; e\,(n+1)\, h_n(S).
\]
\end{theorem}
\begin{proof}
\leanok
\uses{lem:det-growth, lem:det-upper}
If $a_n(S) = 0$ there is nothing to prove. Otherwise chain the growth lemma
at $k = n$ with the upper bound:
$\frac{n^n}{(n+1)^{n+1}}\, a_n(S)\, \Delta_n(S) \le \Delta_{n+1}(S) \le
h_n(S)\, \Delta_n(S)$, and cancel $\Delta_n(S) > 0$. Finally
$(n+1)^{n+1}/n^n = (n+1)(1 + 1/n)^n \le e\,(n+1)$ from $1 + x \le e^x$ at
$x = 1/n$.
\end{proof}

\begin{corollary}[Carl--Pietsch and Mityagin--Henkin, up to the constant $e$]
\label{cor:gelfand-kolmogorov}
\lean{SNumbers.approximationNumber_le_e_mul_sn,
      SNumbers.sn_le_e_mul_tn,
      SNumbers.max_gelfandNumber_kolmogorovNumber_le_e_mul_sn,
      SNumbers.max_gelfandNumber_kolmogorovNumber_le_e_mul_hilbertNumber,
      SNumbers.max_gelfandNumber_kolmogorovNumber_le_e_mul_bernsteinNumber}
\leanok
\uses{thm:maximal-difference, thm:sandwich, def:gelfand-number,
  def:kolmogorov-number, def:bernstein-number}
Let $S \in \mathcal{L}(X, Y)$ and $n \in \mathbb{N}_0$.
\begin{enumerate}
  \item For \emph{any two} $s$-number sequences $s, t$,
    \[
      s_n(S) \;\le\; e\,(n+1)\, t_n(S) :
    \]
    the maximal possible difference between two $s$-number sequences is a factor
    linear in $n$. This is the conjecture of Carl and Pietsch, up to the constant
    $e$.
  \item In particular, for every $s$-number sequence $s$,
    \[
      \max\bigl(c_n(S), d_n(S)\bigr) \;\le\; e\,(n+1)\, s_n(S) ,
    \]
    and taking $s = h$ resp.\ $s = b$ gives
    $\max(c_n, d_n) \le e\,(n+1)\, h_n$ and
    $\max(c_n, d_n) \le e\,(n+1)\, b_n$, the latter being the
    \emph{Mityagin--Henkin conjecture} up to the constant $e$.
\end{enumerate}
\end{corollary}
\begin{proof}
\leanok
\uses{thm:maximal-difference, thm:sandwich}
By the sandwich theorem the Hilbert numbers are the smallest and the
approximation numbers the largest $s$-number sequence, so
Theorem~\ref{thm:maximal-difference} gives
$s_n(S) \le a_n(S) \le e\,(n+1)\, h_n(S) \le e\,(n+1)\, t_n(S)$, which is (1).
For (2) apply (1) with $s := c$ and $s := d$ --- both are $s$-number sequences ---
and take the maximum.
\end{proof}

The factor $n+1$ is order-optimal: the inclusion $I \colon \ell_1 \to
\ell_\infty$ of Section~\ref{sec:l1-linf} satisfies $c_n(I) \ge \tfrac12$ while
$h_n(I) = (n+1)^{-1}$, so no bound $a_n \le \varphi(n)\, h_n$ can hold with
$\varphi(n) = o(n)$ (Corollary~\ref{cor:l1-linf-order-optimal}).

\chapter{Examples}
\label{ch:op-examples}

This chapter computes the $s$-numbers of three concrete families of
operators. The finite-dimensional examples --- the identity between
$\ell_p$-spaces and the diagonal operators --- are the standard test cases
of the theory; the inclusion $\ell_1 \to \ell_\infty$ shows that the linear
factor $n+1$ in the maximal difference theorem
(Theorem~\ref{thm:maximal-difference}) cannot be improved in order. The
chapter is a leaf of the development: it consumes the general theory and
nothing depends on it. Throughout, $\mathbb{K} \in \{\mathbb{R},
\mathbb{C}\}$, and $\ell_p^m$ denotes $\mathbb{K}^m$ with the
$\ell_p$-norm.

\section{The identity \texorpdfstring{$\mathrm{id} : \ell_q^m \to \ell_p^m$}{id : lq to lp}}
\label{sec:ex-identity}

\begin{theorem}[Norm of the identity, $p \le q$]
\label{thm:norm-id-pq}
\lean{SNumbers.norm_idEmbed}
\leanok
For $1 \le p \le q < \infty$ and $m \ge 1$,
\[
  \bigl\| \mathrm{id} : \ell_q^m \to \ell_p^m \bigr\|
  \;=\; m^{\,1/p - 1/q}.
\]
\end{theorem}
\begin{proof}
\leanok
The upper bound is the power-mean (H\"older) inequality
$\|x\|_p \le m^{1/p-1/q} \|x\|_q$; the lower bound tests the all-ones
vector, whose $\ell_q$-norm is $m^{1/q}$ and whose $\ell_p$-norm is
$m^{1/p}$. Formalised in \texttt{Identity.lean}.
\end{proof}

\begin{theorem}[$s$-numbers of the identity, $p \le q$]
\label{thm:sn-id-pq}
\lean{SNumbers.approximationNumber_idEmbed_eq}
\leanok
\uses{def:approximation-number, thm:norm-id-pq, thm:sandwich}
For $1 \le p \le q < \infty$ and $n < m$,
\[
  a_n\bigl( \mathrm{id} : \ell_q^m \to \ell_p^m \bigr)
  \;=\; (m - n)^{\,1/p - 1/q}.
\]
By the sandwich theorem, every $s$-number sequence obeys
$s_n(\mathrm{id}) \le (m-n)^{1/p-1/q}$.
\end{theorem}
\begin{proof}
\leanok
\uses{thm:norm-id-pq}
For the upper bound, truncate to the first $n$ coordinates: the residual is
the identity on the remaining $m - n$ coordinates, of norm
$(m-n)^{1/p-1/q}$ by (the proof of) Theorem~\ref{thm:norm-id-pq}. For the
lower bound, a rank-$\le n$ operator $L$ has kernel of dimension at least
$m - n$, and a flatness (pigeonhole) argument produces a nonzero $x$ in
the kernel with $\|x\|_p \ge (m-n)^{1/p-1/q} \|x\|_q$; on such an $x$ the
error $\mathrm{id} - L$ acts as the identity, so
$\|\mathrm{id} - L\| \ge (m-n)^{1/p-1/q}$.
\end{proof}

\begin{proposition}[Strict normalisation on $\ell_p^m$]
\label{prop:sn-id-p}
\lean{SNumbers.sn_id_piLp}
\leanok
\uses{def:strict-s-number-sequence}
For every strict $s$-number sequence $s$, every $1 \le p \le \infty$ and
$n < m$, we have $s_n(\mathrm{id}_{\ell_p^m}) = 1$.
\end{proposition}
\begin{proof}
\leanok
This is exactly the strict normalisation (S5$'$) applied to
$X = \ell_p^m$, whose dimension is $m > n$.
\end{proof}

\section{Diagonal operators}
\label{sec:ex-diagonal}

For $\sigma = (\sigma_0, \dots, \sigma_{m-1}) \in \mathbb{K}^m$ let
$D_\sigma$ denote the diagonal operator
\[
  D_\sigma : (x_i)_i \longmapsto (\sigma_i \, x_i)_i .
\]
Throughout this section, $\sigma$ is \emph{non-increasing in modulus},
$\|\sigma_0\| \ge \|\sigma_1\| \ge \dots \ge \|\sigma_{m-1}\|$, and
$n < m$.

\begin{theorem}[Diagonal operators, same exponent]
\label{thm:diag-same}
\lean{SNumbers.norm_DiagCLM, SNumbers.sn_DiagCLM_eq,
      SNumbers.hilbertNumber_DiagCLM_le}
\leanok
\uses{def:strict-s-number-sequence, thm:sandwich, prop:sn-id-p}
Let $1 \le p \le \infty$ and $D_\sigma : \ell_p^m \to \ell_p^m$. Then
$\|D_\sigma\| = \max_i \|\sigma_i\|$, and for every \emph{strict}
$s$-number sequence $s$ (in particular for $a_n$, $b_n$, $c_n$, $d_n$),
\[
  s_n(D_\sigma) \;=\; \|\sigma_n\| .
\]
The Hilbert numbers satisfy $h_n(D_\sigma) \le \|\sigma_n\|$; equality can
fail for $p \neq 2$.
\end{theorem}
\begin{proof}
\leanok
\uses{prop:sn-id-p}
The norm is attained at the coordinate carrying $\max_i \|\sigma_i\| =
\|\sigma_0\|$. Upper bound: truncating $\sigma$ to its first $n$ entries
gives a rank-$\le n$ approximant whose residual is the diagonal operator
with entries $(\sigma_k)_{k \ge n}$, of norm $\|\sigma_n\|$; hence
$s_n(D_\sigma) \le a_n(D_\sigma) \le \|\sigma_n\|$ for \emph{every}
$s$-number sequence. Lower bound for strict $s$: since
$\|\sigma_k\| \ge \|\sigma_n\|$ for $k \le n$, the identity of
$\ell_p^{n+1}$ factors as $\mathrm{id} = B \circ D_\sigma \circ A$ with the
coordinate embedding $A$ (a contraction) and $B$ the coordinate projection
followed by the diagonal operator with entries $\sigma_k^{-1}$, of norm
$\le \|\sigma_n\|^{-1}$; then (S3) and the strict normalisation
(Proposition~\ref{prop:sn-id-p}) give
$1 = s_n(\mathrm{id}) \le \|\sigma_n\|^{-1} s_n(D_\sigma)$. Formalised in
\texttt{DiagonalMatrices.lean}.
\end{proof}

\begin{theorem}[Diagonal operators, mixed exponents $p < q$]
\label{thm:diag-mixed}
\lean{SNumbers.norm_DiagCLMpq_eq, SNumbers.approximationNumber_DiagCLMpq_eq,
      SNumbers.sn_DiagCLMpq_le}
\leanok
\uses{def:approximation-number, thm:sandwich, thm:sn-id-pq}
Let $1 \le p < q < \infty$ and define $r$ by
$\tfrac1r = \tfrac1p - \tfrac1q$. For $D_\sigma : \ell_q^m \to \ell_p^m$,
\[
  a_n(D_\sigma)
  \;=\; \Bigl( \sum_{k = n}^{m-1} \|\sigma_k\|^r \Bigr)^{1/r} ,
  \qquad\text{in particular}\qquad
  \|D_\sigma\| = \|\sigma\|_{\ell_r} .
\]
The upper bound $s_n(D_\sigma) \le \bigl(\sum_{k \ge n}
\|\sigma_k\|^r\bigr)^{1/r}$ holds for every $s$-number sequence and needs
no ordering of the diagonal.
\end{theorem}
\begin{proof}
\leanok
\uses{thm:sn-id-pq}
Upper bound: truncate to the first $n$ coordinates; the residual diagonal
has norm equal to the $\ell_r$-norm of its entries, by H\"older's
inequality applied with the exponent pair $(q/r', \dots)$ and attained at
the explicitly weighted vector $x_k = \|\sigma_k\|^{r/q}$. Lower bound:
for a rank-$\le n$ operator $L$, a weighted flatness argument on
$\ker L$ (the weighted analogue of the pigeonhole step in
Theorem~\ref{thm:sn-id-pq}) produces a kernel vector on which
$\mathrm{id} - L$ retains the full tail $\ell_r$-norm. The case $n = 0$,
$L = 0$ yields the operator norm.
\end{proof}

\begin{theorem}[Diagonal operators, reverse regime $q \le p$]
\label{thm:diag-reverse}
\lean{SNumbers.norm_DiagCLMpq_iSup, SNumbers.sn_DiagCLMpq_le_of_exponent_ge}
\leanok
\uses{thm:diag-same, thm:sandwich}
Let $1 \le q \le p \le \infty$ (including $p = \infty$). For
$D_\sigma : \ell_q^m \to \ell_p^m$,
\[
  \|D_\sigma\| = \max_i \|\sigma_i\| ,
  \qquad
  s_n(D_\sigma) \le \|\sigma_n\|
  \quad\text{for every $s$-number sequence $s$.}
\]
\end{theorem}
\begin{proof}
\leanok
\uses{thm:diag-same}
Since $q \le p$, the identity $\ell_q^m \to \ell_p^m$ is a contraction, and
$D_\sigma$ factors as this contraction after the same-exponent diagonal
operator on $\ell_q^m$; the ideal property (S3) transfers both bounds
from Theorem~\ref{thm:diag-same}. The lower bound for the norm again tests
a single coordinate.
\end{proof}

For $q < p$ the upper bound $s_n(D_\sigma) \le \|\sigma_n\|$ need not be
sharp: the exact values have no elementary closed form in this regime.

\section{The inclusion \texorpdfstring{$\ell_1 \to \ell_\infty$}{l1 to linfty}}
\label{sec:l1-linf}

This section computes both sides of the maximal difference theorem for the
natural inclusion
\[
  I \colon \ell_1 \to \ell_\infty, \qquad (x_j)_j \mapsto (x_j)_j ,
\]
a contraction which is \emph{not} compact. Everything here is formalised in
\texttt{IdentityL1Linfty.lean}, with the two little

\subsection{Sign averaging and two little Grothendieck bounds}
\label{sec:l1-linf-grothendieck}

\begin{lemma}[Sign averaging]
\label{lem:sign-averaging}
\lean{SNumbers.sum_norm_sq_le_sq_of_signedSum_le}
\leanok
Let $w_j$ ($j \in J$, $J$ finite) be vectors in an inner product space and
$M \in \mathbb{R}$. If $\|\sum_{j \in J} \varepsilon_j w_j\| \le M$ for every
choice of signs $\varepsilon \in \{\pm 1\}^J$, then
$\sum_{j \in J} \|w_j\|^2 \le M^2$.
\end{lemma}
\begin{proof}
\leanok
Summing over all $2^{|J|}$ sign patterns, the cross terms cancel:
\[
  \sum_{\varepsilon \in \{\pm 1\}^J} \Bigl\| \sum_{j \in J} \varepsilon_j w_j \Bigr\|^2
  = 2^{|J|} \sum_{j \in J} \|w_j\|^2 ,
\]
proved by induction on $J$, pairing the two extensions of each sign pattern to a
new index by the parallelogram law. Each summand on the left is at most $M^2$,
and there are $2^{|J|}$ of them.
\end{proof}

\begin{lemma}[Little Grothendieck bounds, constant one]
\label{lem:little-grothendieck}
\lean{SNumbers.sum_norm_sq_apply_single_le,
      SNumbers.sum_norm_sq_row_le}
\leanok
\uses{lem:sign-averaging}
Let $H$ be a Hilbert space and $J$ a finite set of indices.
\begin{enumerate}
  \item For every $B \in \mathcal{L}(\ell_\infty, H)$,
    $\displaystyle \sum_{j \in J} \|B e_j\|^2 \le \|B\|^2$.
  \item For every $A \in \mathcal{L}(H, \ell_1)$ with rows $w_j \in H$, i.e.
    $\langle w_j, x\rangle = (A x)_j$ for all $x$,
    $\displaystyle \sum_{j \in J} \|w_j\|^2 \le \|A\|^2$.
\end{enumerate}
In Hilbert--Schmidt language: both operators are Hilbert--Schmidt with
$\|\cdot\|_{HS} \le \|\cdot\|$.
\end{lemma}
\begin{proof}
\leanok
\uses{lem:sign-averaging}
(1) The vectors $\sum_{j \in J} \varepsilon_j e_j$ all have $\ell_\infty$-norm
$1$, so $\|\sum_j \varepsilon_j B e_j\| \le \|B\|$ and
Lemma~\ref{lem:sign-averaging} applies with $M = \|B\|$.

(2) For a signed sum $z := \sum_{j \in J}\varepsilon_j w_j$ the rows reproduce
the signs on the coordinates of $A z$:
\[
  \|z\|^2 = \operatorname{Re}\langle z, z\rangle
          = \operatorname{Re} \sum_{j \in J} \varepsilon_j (A z)_j
          \le \sum_{j \in J} |(A z)_j| \le \|A z\|_1 \le \|A\|\,\|z\| ,
\]
so $\|z\| \le \|A\|$, and Lemma~\ref{lem:sign-averaging} applies with
$M = \|A\|$. This is the little Grothendieck bound for the dual operator
$\ell_\infty = \ell_1^{\,\prime} \to H$, formulated through the rows so that no
adjoint has to be constructed.
\end{proof}

\subsection{The Gelfand numbers of \texorpdfstring{$I$}{I}}
\label{sec:l1-linf-gelfand}

\begin{proposition}[$\tfrac12 \le c_n(I) \le 1$]
\label{prop:l1-linf-gelfand}
\lean{SNumbers.L1Linf.gelfand_le_one,
      SNumbers.L1Linf.half_le_gelfandNumber}
\leanok
\uses{def:gelfand-number}
For all $n$, $\tfrac12 \le c_n(I) \le 1$.
\end{proposition}
\begin{proof}
\leanok
\uses{def:gelfand-number}
The upper bound is $c_n(I) \le \|I\| \le 1$. For the lower bound let
$N \subseteq \ell_1$ be closed of codimension $\le n$. Then $\ell_1/N$ is
finite dimensional, so among the classes $[e_i]$ of the unit vectors two are
arbitrarily close (Bolzano--Weierstrass). Lifting the difference gives
$x \in N$ close to $e_i - e_{i'}$, with $\|x\|_1 \approx 2$ but
$\|I x\|_\infty \approx 1$; hence $\|I|_N\| \ge \tfrac12 - \varepsilon$ for
every $\varepsilon > 0$.
\end{proof}

\subsection{The Hilbert numbers of \texorpdfstring{$I$}{I}}
\label{sec:l1-linf-hilbert}

\begin{proposition}[Lower bound]
\label{prop:l1-linf-hilbert-lower}
\lean{SNumbers.L1Linf.one_div_le_hilbertNumber}
\leanok
\uses{def:hilbert-number, prop:hilbert-is-snumber}
$h_n(I) \ge (n+1)^{-1}$ for all $n$.
\end{proposition}
\begin{proof}
\leanok
\uses{def:hilbert-number, prop:hilbert-is-snumber}
The identity of $\ell_2^{\,n+1}$ factors as
$\mathrm{id}_{\ell_2^{\,n+1}} = \mathrm{proj}_\infty \circ I \circ \mathrm{emb}_1$
with $\|\mathrm{emb}_1\|, \|\mathrm{proj}_\infty\| \le \sqrt{n+1}$. By the ideal
property of the Hilbert numbers and their normalisation
$h_n(\mathrm{id}_{\ell_2^{\,n+1}}) \ge 1$,
\[
  1 \le \|\mathrm{proj}_\infty\|\, h_n(I)\, \|\mathrm{emb}_1\| \le (n+1)\, h_n(I) .
\]
\end{proof}

\begin{theorem}[Upper bound]
\label{thm:l1-linf-hilbert-upper}
\lean{SNumbers.L1Linf.hilbertNumber_le_one_div}
\leanok
\uses{def:hilbert-number, lem:little-grothendieck, lem:bessel-rows,
  thm:hs-product-bound, thm:approximable-implies-compact}
$h_n(I) \le (n+1)^{-1}$ for all $n$.
\end{theorem}
\begin{proof}
\leanok
\uses{lem:little-grothendieck, lem:bessel-rows, thm:hs-product-bound,
  thm:approximable-implies-compact}
Let $A \in \mathcal{L}(\ell_2, \ell_1)$ and $B \in \mathcal{L}(\ell_\infty,
\ell_2)$ be non-zero; we must bound $a_n(B I A) \le \|A\|\,\|B\|/(n+1)$.
Factorising $I = J_2 J_1$ with the norm-one inclusions
$J_1 \colon \ell_1 \to \ell_2$ and $J_2 \colon \ell_2 \to \ell_\infty$, we obtain
$T := B I A = T_2 T_1$ with $T_1 := J_1 A$ and $T_2 := B J_2$ on $\ell_2$.
Applying Lemma~\ref{lem:little-grothendieck} to $B$ and to the rows of $A$,
\[
  \|T_2\|_{HS}^2 = \sum_j \|B e_j\|^2 \le \|B\|^2
  \quad\text{and}\quad
  \|T_1\|_{HS}^2 = \sum_j \|\mathrm{row}_j(A)\|^2 \le \|A\|^2 ,
\]
which by Lemma~\ref{lem:bessel-rows} gives the two hypotheses of
Theorem~\ref{thm:hs-product-bound} along the singular vectors of $T$. That
theorem yields $(n+1)\, a_n(T) \le \|A\|\,\|B\|$ \emph{provided $T$ is compact}.

For compactness, replace $A$ by its coordinate truncation $A_m$, of rank
$\le m$ and with $\|A_m\| \le \|A\|$; then $B I A_m$ has finite rank, hence is
compact (Theorem~\ref{thm:approximable-implies-compact}), and the estimate above
applies to it. Since $\sum_j \|\mathrm{row}_j(A)\|^2 < \infty$ we have
$\|\mathrm{row}_j(A)\| \to 0$, so $\|I(A - A_m)\| \to 0$, and subadditivity of the
approximation numbers,
\[
  a_n(B I A) \le a_n(B I A_m) + \|B\|\,\|I(A - A_m)\| ,
\]
lets $m \to \infty$. Dividing by $\|A\|\,\|B\|$ and taking the supremum over all
admissible $A, B$ gives $h_n(I) \le (n+1)^{-1}$.
\end{proof}

\begin{corollary}[$h_n(I) = (n+1)^{-1}$]
\label{cor:l1-linf-hilbert-eq}
\lean{SNumbers.L1Linf.hilbertNumber_eq_one_div}
\leanok
\uses{thm:l1-linf-hilbert-upper, prop:l1-linf-hilbert-lower}
$h_n(I) = (n+1)^{-1}$ for all $n$.
\end{corollary}
\begin{proof}
\leanok
\uses{thm:l1-linf-hilbert-upper, prop:l1-linf-hilbert-lower}
Combine the two bounds.
\end{proof}

\subsection{Order-optimality of the factor \texorpdfstring{$n+1$}{n+1}}
\label{sec:l1-linf-optimal}

\begin{corollary}[The factor $n+1$ is order-optimal]
\label{cor:l1-linf-order-optimal}
\lean{SNumbers.L1Linf.mul_hilbertNumber_le_gelfandNumber}
\leanok
\uses{cor:l1-linf-hilbert-eq, prop:l1-linf-gelfand, thm:maximal-difference}
For all $n$,
\[
  \frac{n+1}{2}\, h_n(I) \;\le\; c_n(I) \;\le\; a_n(I) ,
\]
whereas the maximal difference theorem gives $a_n \le e\,(n+1)\,h_n$ in general.
Hence no bound $a_n \le \varphi(n)\,h_n$ can hold with $\varphi(n) = o(n)$.
\end{corollary}
\begin{proof}
\leanok
\uses{cor:l1-linf-hilbert-eq, prop:l1-linf-gelfand}
By Corollary~\ref{cor:l1-linf-hilbert-eq}, $\tfrac{n+1}{2}\, h_n(I) = \tfrac12$,
which is at most $c_n(I)$ by Proposition~\ref{prop:l1-linf-gelfand}.
\end{proof}

\chapter{Injective and surjective \texorpdfstring{$s$}{s}-numbers}
\label{ch:injectivity}

Two further structural properties of an $s$-number sequence, from Pietsch's
theory, single out the Gelfand and Kolmogorov numbers. They concern the
behaviour of $s_n$ under composition with two distinguished classes of maps.
Everything in this chapter is fully formalised in
\texttt{Injectivity.lean}.

\begin{definition}[Metric injection]
\label{def:metric-injection}
\lean{SNumbers.IsMetricInjection}
\leanok
A \emph{metric injection} is a continuous linear map $J : Y \to Z$ that
preserves norms: $\|J y\| = \|y\|$ for all $y$ (an isometric embedding). In
particular $\|J\| \le 1$.
\end{definition}

\begin{definition}[Metric surjection]
\label{def:metric-surjection}
\lean{SNumbers.IsMetricSurjection}
\leanok
A \emph{metric surjection} is a continuous linear map $Q : W \to X$ that is a
contraction ($\|Q\| \le 1$) and maps the open unit ball \emph{onto} the open
unit ball: every $x$ has preimages of norm arbitrarily close to $\|x\|$. This
is the quotient-map condition $\|x\| = \inf\{\|w\| : Q w = x\}$.
\end{definition}

\begin{lemma}[Norms are preserved under the two compositions]
\label{lem:metric-norm-comp}
\lean{SNumbers.IsMetricInjection.norm_comp,
      SNumbers.IsMetricSurjection.norm_comp}
\leanok
\uses{def:metric-injection, def:metric-surjection}
Post-composition with a metric injection and pre-composition with a metric
surjection both leave the operator norm unchanged:
$\|J \circ T\| = \|T\|$ and $\|T \circ Q\| = \|T\|$.
\end{lemma}
\begin{proof}
\leanok
For the injection, $\|(J \circ T)x\| = \|J(Tx)\| = \|Tx\|$ pointwise, so the
two operator norms agree. For the surjection, $\|T \circ Q\| \le \|T\|$ since
$\|Q\| \le 1$; conversely, lifting each $x$ to a preimage $w$ with
$\|w\| \approx \|x\|$ gives $\|Tx\| = \|(T\circ Q)w\| \le \|T\circ Q\|\,\|w\|$,
and letting the slack tend to $0$ yields $\|T\| \le \|T \circ Q\|$.
\end{proof}

\begin{definition}[Injective / surjective $s$-number sequence]
\label{def:injective-surjective}
\lean{SNumbers.Injective, SNumbers.Surjective}
\leanok
\uses{def:s-number-sequence, def:metric-injection, def:metric-surjection}
An $s$-number sequence $s$ is \emph{injective} if post-composition with any
metric injection leaves it unchanged, $s_n(J \circ S) = s_n(S)$; it is
\emph{surjective} if pre-composition with any metric surjection leaves it
unchanged, $s_n(S \circ Q) = s_n(S)$.
\end{definition}

\begin{theorem}[The Gelfand numbers are injective]
\label{thm:gelfand-injective}
\lean{SNumbers.injective_gelfandNumber}
\leanok
\uses{def:injective-surjective, def:gelfand-number, lem:metric-norm-comp}
The Gelfand numbers $c_n$ form an injective $s$-number sequence:
$c_n(J \circ S) = c_n(S)$ for every metric injection $J$.
\end{theorem}
\begin{proof}
\leanok
\uses{lem:metric-norm-comp}
The infimum defining $c_n$ ranges over restriction norms $\|S|_M\|$, and
$\|(J \circ S)|_M\| = \|J \circ (S|_M)\| = \|S|_M\|$ by
Lemma~\ref{lem:metric-norm-comp}. The two infimum sets therefore coincide
term by term, so the Gelfand numbers agree.
\end{proof}

\begin{theorem}[The Kolmogorov numbers are surjective]
\label{thm:kolmogorov-surjective}
\lean{SNumbers.surjective_kolmogorovNumber}
\leanok
\uses{def:injective-surjective, def:kolmogorov-number, lem:metric-norm-comp}
The Kolmogorov numbers $d_n$ form a surjective $s$-number sequence:
$d_n(S \circ Q) = d_n(S)$ for every metric surjection $Q$.
\end{theorem}
\begin{proof}
\leanok
\uses{lem:metric-norm-comp}
The infimum defining $d_n$ ranges over quotient norms
$\|\pi_V \circ S\|$, and
$\|\pi_V \circ (S \circ Q)\| = \|(\pi_V \circ S) \circ Q\| = \|\pi_V \circ S\|$
by Lemma~\ref{lem:metric-norm-comp}. The two infimum sets coincide term by
term, so the Kolmogorov numbers agree.
\end{proof}

\chapter{Entropy numbers}
\label{ch:entropy}

The entropy numbers quantify the compactness of an operator directly: they
measure how many balls of a given radius are needed to cover the image of
the unit ball. They are \emph{not} $s$-numbers --- the rank axiom (S4) and
the normalisation (S5) both fail --- but they satisfy the remaining
properties, in the strong additive and multiplicative form below; such
sequences are called \emph{pseudo-$s$-number} sequences (Pietsch,
\emph{Operator ideals}, 12.1.1). Everything in this chapter is formalised
in \texttt{Entropy.lean}, over a non-trivially normed field
$\mathbb{K}$ unless stated otherwise.

\begin{definition}[Entropy number]
\label{def:entropy-number}
\lean{SNumbers.entropyNumber}
\leanok
For $S \in \mathcal{L}(X, Y)$ and $n \in \mathbb{N}_0$,
\[
  e_n(S) := \inf \bigl\{ \varepsilon > 0 : S(B_X) \text{ is covered by at
  most } 2^n \text{ closed $\varepsilon$-balls in } Y \bigr\},
\]
where $B_X$ is the closed unit ball of $X$ and the centres of the covering
balls are arbitrary points of $Y$. (The indexing is $0$-based, so $e_0$
allows a single ball.)
\end{definition}

\begin{proposition}[Basic properties]
\label{prop:entropy-basic}
\lean{SNumbers.entropyNumber_nonneg, SNumbers.entropyNumber_antitone,
      SNumbers.entropyNumber_le_norm}
\leanok
\uses{def:entropy-number}
For every $S \in \mathcal{L}(X, Y)$ and $n \in \mathbb{N}_0$,
\[
  0 \;\le\; e_{n+1}(S) \;\le\; e_n(S) \;\le\; \|S\| .
\]
\end{proposition}
\begin{proof}
\leanok
Nonnegativity and monotonicity are immediate from the definition (any
cover by $2^n$ balls is a cover by $2^{n+1}$ balls). For the norm bound,
a single ball of radius $\|S\| + \delta$ centred at the origin covers
$S(B_X)$, for every $\delta > 0$.
\end{proof}

\begin{theorem}[Additivity; (S2)]
\label{thm:entropy-additive}
\lean{SNumbers.entropyNumber_additive, SNumbers.entropyNumber_add_le}
\leanok
\uses{def:entropy-number, prop:entropy-basic}
For all $S, T \in \mathcal{L}(X, Y)$ and $n, m \in \mathbb{N}_0$,
\[
  e_{n+m}(S + T) \;\le\; e_n(S) + e_m(T) .
\]
In particular, taking $m = 0$ and using $e_0(T) \le \|T\|$ recovers the
$s$-number axiom (S2): $e_n(S + T) \le e_n(S) + \|T\|$.
\end{theorem}
\begin{proof}
\leanok
\uses{prop:entropy-basic}
If $S(B_X)$ is covered by $2^n$ balls of radius $\varepsilon$ with centres
$y_i$ and $T(B_X)$ by $2^m$ balls of radius $\delta$ with centres $z_j$,
then the $2^{n+m}$ balls of radius $\varepsilon + \delta$ centred at the
sums $y_i + z_j$ cover $(S+T)(B_X)$.
\end{proof}

\begin{theorem}[Multiplicativity; (S3)]
\label{thm:entropy-multiplicative}
\lean{SNumbers.entropyNumber_multiplicative, SNumbers.entropyNumber_comp_comp_le}
\leanok
\uses{def:entropy-number, prop:entropy-basic}
Let the norm values of $\mathbb{K}$ be dense in $[0, \infty)$ (as for
$\mathbb{R}$ and $\mathbb{C}$). Then for all $S \in \mathcal{L}(X, Y)$,
$B \in \mathcal{L}(Y, Z)$ and $n, m \in \mathbb{N}_0$,
\[
  e_{n+m}(B \circ S) \;\le\; e_n(B)\, e_m(S) .
\]
In particular the $s$-number axiom (S3) holds:
$e_n(B \circ S \circ A) \le \|B\| \, e_n(S) \, \|A\|$.
\end{theorem}
\begin{proof}
\leanok
\uses{prop:entropy-basic}
Cover $S(B_X)$ by $2^m$ balls of radius $\delta$ with centres $y_i$. Each
point of $S(B_X)$ in the ball around $y_i$ differs from $y_i$ by a vector
$u$ with $\|u\| \le \delta$, which is rescaled as $u = c\,w$ with
$\|w\| \le 1$ and $\|c\|$ close to $\delta$ --- this is where density of
the norm values enters. Covering $B(B_Y)$ by $2^n$ balls of radius
$\varepsilon$ then covers each $B(u)$ by $2^n$ balls of radius
$\|c\|\,\varepsilon$, and combining the two covers yields $2^{n+m}$ balls
of radius close to $\varepsilon\delta$ around $B(S(B_X))$. The corollary
takes one index to be $0$ and uses $e_0 \le \|\cdot\|$ twice.
\end{proof}

\begin{remark}[Entropy numbers are not $s$-numbers]
\label{rem:entropy-not-snumber}
\uses{def:entropy-number, def:s-number-sequence}
Two of Pietsch's axioms fail, and one is not formalised:
\begin{itemize}
  \item (S4) fails: if $e_n(S) = 0$ then $S(B_X)$ is covered by $2^n$
    balls of arbitrarily small radius, hence has at most $2^n$ points ---
    impossible for $S \neq 0$. So $e_n(S) > 0$ even for finite-rank $S$.
  \item (S5) fails for $n \ge 2$: covering the Euclidean unit ball of
    $\mathbb{R}^{n+1}$ by balls of radius $< 1$ takes only $n + 2$ of
    them, and $2^n \ge n + 2$ from $n = 2$ on, so
    $e_n(\mathrm{id}_{\ell_2^{n+1}}) < 1$.
  \item (S1a), the equality $e_0(S) = \|S\|$, holds over $\mathbb{R}$ and
    $\mathbb{C}$ by symmetry of $S(B_X)$, but fails over general normed
    fields; only the inequality $e_0(S) \le \|S\|$ is formalised, which is
    all the results above need.
\end{itemize}
\end{remark}

\begin{theorem}[Entropy numbers measure compactness]
\label{thm:entropy-compact}
\lean{SNumbers.isCompactOperator_iff_tendsto_entropyNumber,
      SNumbers.tendsto_entropyNumber_iff_totallyBounded}
\leanok
\uses{def:entropy-number}
For every $S \in \mathcal{L}(X, Y)$, the sequence $e_n(S)$ tends to $0$ if
and only if $S(B_X)$ is totally bounded. If $Y$ is complete, then
\[
  S \text{ is compact} \iff e_n(S) \xrightarrow{\,n \to \infty\,} 0 ,
\]
where the forward implication needs no completeness.
\end{theorem}
\begin{proof}
\leanok
The first equivalence matches the definitions: a finite $\varepsilon$-net
of $S(B_X)$ of size $N$ is a cover by $2^n$ balls once $2^n \ge N$, and
conversely. If $S$ is compact, the closure of $S(B_X)$ is compact, hence
totally bounded; if conversely $S(B_X)$ is totally bounded and $Y$ is
complete, its closure is compact, which is the definition of a compact
operator.
\end{proof}
